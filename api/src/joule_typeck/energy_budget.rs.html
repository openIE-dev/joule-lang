1//! Compile-time Energy Budget Estimation
2//!
3//! This module implements static analysis for estimating energy consumption
4//! of HIR functions. It uses the energy cost model from `joule-energy` to
5//! compute approximate energy costs based on the operations in the code.
6//!
7//! # Key Features
8//!
9//! - Walks HIR expressions and accumulates energy costs
10//! - Handles loops with configurable iteration bounds
11//! - Tracks confidence levels for estimates (lower for unbounded loops)
12//! - Compares estimates against declared budgets
13//!
14//! # Example
15//!
16//! ```ignore
17//! use joule_typeck::EnergyEstimator;
18//! use joule_hir::Function;
19//!
20//! let estimator = EnergyEstimator::default();
21//! let estimate = estimator.estimate_function(&func);
22//!
23//! if let Some(violation) = estimator.check_budget(&func) {
24//!     println!("warning: {}", violation.message);
25//! }
26//! ```

35/// Energy estimator for HIR functions.
36///
37/// Uses the calibrated energy cost model to estimate the energy consumption
38/// of code at compile time.
39#[derive(Debug, Clone)]
40pub struct EnergyEstimator {
41    /// The energy cost model with calibrated instruction costs
42    cost_model: EnergyCostModel,
43    /// Default iteration count for loops with unknown bounds
44    default_loop_iterations: u64,
45    /// Maximum loop iterations to consider (to prevent infinite estimates)
46    max_loop_iterations: u64,
47    /// Thermal factor to use for estimation (0.0 = cold, 1.0 = hot)
48    thermal_factor: f64,
49}
50
51impl EnergyEstimator {
52    /// Create a new energy estimator with the default calibrated cost model
53    pub fn new() -> Self {
54        Self {
55            cost_model: EnergyCostModel::default_calibrated(),
56            default_loop_iterations: 100,
57            max_loop_iterations: 10_000,
58            thermal_factor: 0.3, // Assume nominal thermal state
59        }
60    }

62    /// Create an energy estimator with a custom cost model
63    pub fn with_cost_model(cost_model: EnergyCostModel) -> Self {
64        Self {
65            cost_model,
66            default_loop_iterations: 100,
67            max_loop_iterations: 10_000,
68            thermal_factor: 0.3,

88    fn effective_loop_iterations(&self) -> u64 {
89        self.default_loop_iterations.min(self.max_loop_iterations)

92    /// Reference CPU frequency for power estimation (3.0 GHz)
93    const REFERENCE_FREQUENCY_GHZ: f64 = 3.0;
94    /// Thermal resistance for temperature estimation (K/W)
95    /// Typical value for a modern CPU with standard cooling
96    const THERMAL_RESISTANCE_KW: f64 = 0.4;
97
98    /// Get the energy cost for an instruction type
99    fn instruction_cost(&self, instr: InstructionType) -> f64 {
100        self.cost_model
101            .get(&instr)
102            .map(|e| e.actual_cost(self.thermal_factor))
103            .unwrap_or(0.1) // Small default cost for unknown instructions
104    }

106    /// Estimate the energy cost of a function
107    ///
108    /// Returns (estimated_joules, confidence)
109    /// - estimated_joules: Total estimated energy in joules
110    /// - confidence: 0.0 to 1.0, lower if there are unbounded loops
111    pub fn estimate_function(&self, func: &Function) -> (f64, f64) {
112        let mut ctx = EstimationContext::new();
113        self.estimate_block(&func.body, &mut ctx);
114
115        // Convert picojoules to joules
116        let joules = ctx.total_cost_pj / 1e12;
117
118        (joules, ctx.confidence)
119    }
120
121    /// Estimate the average power consumption of a function in watts.
122    ///
123    /// Power = Energy / Time. Execution time is derived from the total energy
124    /// and the average picojoules-per-cycle ratio from the cost model.
125    /// The ratio is calibrated from the instruction mix's energy/cycle profile.
126    pub fn estimate_function_power(&self, func: &Function) -> (f64, f64) {
127        let (estimated_joules, confidence) = self.estimate_function(func);
128
129        // Derive execution time from energy using average energy-per-cycle.
130        // Average pJ/cycle across typical instruction mix:
131        //   IntAdd: 0.05 pJ / 1 cycle = 0.05
132        //   FloatMul: 0.35 pJ / 3 cycles = 0.117
133        //   MemLoadL1: 0.5 pJ / 4 cycles = 0.125
134        //   MemLoadDRAM: 200 pJ / 200 cycles = 1.0
135        // Weighted average for mixed workloads: ~0.15 pJ/cycle
136        let avg_pj_per_cycle = 0.15;
137        let total_cost_pj = estimated_joules * 1e12;
138        let estimated_cycles = (total_cost_pj / avg_pj_per_cycle).max(1.0);
139        let execution_time_seconds = estimated_cycles / (Self::REFERENCE_FREQUENCY_GHZ * 1e9);
140        let watts = estimated_joules / execution_time_seconds;
141
142        (watts, confidence)
143    }

145    /// Estimate the temperature delta caused by executing a function.
146    ///
147    /// Uses a simplified thermal model: Delta_T = Power * Thermal_Resistance.
148    /// The thermal resistance accounts for the CPU die's ability to dissipate heat.
149    /// For sustained workloads this gives a reasonable upper bound on heating.
150    pub fn estimate_function_temp_delta(&self, func: &Function) -> (f64, f64) {
151        let (watts, confidence) = self.estimate_function_power(func);
152        let temp_delta = watts * Self::THERMAL_RESISTANCE_KW;
153        (temp_delta, confidence)
154    }

156    /// Estimate the energy cost of a block
157    fn estimate_block(&self, block: &Block, ctx: &mut EstimationContext) {
158        for stmt in &block.statements {
159            self.estimate_statement(stmt, ctx);
160        }
161
162        if let Some(expr) = &block.expr {
163            self.estimate_expr(expr, ctx);
164        }
165    }

167    /// Estimate the energy cost of a statement
168    fn estimate_statement(&self, stmt: &Statement, ctx: &mut EstimationContext) {
169        match stmt {
170            Statement::Let { init, .. } => {
171                if let Some(init_expr) = init {
172                    self.estimate_expr(init_expr, ctx);
173                }
174                // Memory store for the variable binding
175                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1));
176            }
177            Statement::Expr(expr) => {
178                self.estimate_expr(expr, ctx);
179            }
180        }
181    }

183    /// Estimate the energy cost of an expression
184    fn estimate_expr(&self, expr: &Expr, ctx: &mut EstimationContext) {
185        match &expr.kind {
186            ExprKind::Literal(_) => {
187                // Literals are essentially free (just a register load)
188                ctx.add_cost(0.01);
189            }
190
191            ExprKind::Var { .. } => {
192                // Variable access - assume L1 cache hit
193                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1));
194            }
195
196            ExprKind::Binary { op, left, right } => {
197                self.estimate_expr(left, ctx);
198                self.estimate_expr(right, ctx);
199                ctx.add_cost(self.binary_op_cost(*op, &left.ty));
200            }
201
202            ExprKind::Unary { op, expr: inner } => {
203                self.estimate_expr(inner, ctx);
204                ctx.add_cost(self.unary_op_cost(*op, &inner.ty));
205            }
206
207            ExprKind::Call { func: callee, args } => {
208                for arg in args {
209                    self.estimate_expr(arg, ctx);
210                }
211                // Function call overhead: branch + stack manipulation
212                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken));
213                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1) * 2.0);
214
215                // Transitive: add callee's estimated energy if available
216                if let Some(ref callee_map) = ctx.callee_costs {
217                    if let Some(name) = Self::extract_callee_name(callee) {
218                        if let Some(&callee_joules) = callee_map.get(&name) {
219                            let callee_pj = callee_joules * 1e12;
220                            ctx.add_cost(callee_pj);
221                            ctx.record_callee(name, callee_joules);
222                        }
223                    }
224                }
225            }
226
227            ExprKind::MethodCall {
228                receiver,
229                method,
230                args,
231            } => {
232                self.estimate_expr(receiver, ctx);
233                for arg in args {
234                    self.estimate_expr(arg, ctx);
235                }
236                // Method call has slightly more overhead than direct call
237                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken));
238                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1) * 3.0);
239
240                // Transitive: add callee's estimated energy if available
241                if let Some(ref callee_map) = ctx.callee_costs {
242                    let method_name = method.as_str();
243                    if let Some(&callee_joules) = callee_map.get(&method_name) {
244                        let callee_pj = callee_joules * 1e12;
245                        ctx.add_cost(callee_pj);
246                        ctx.record_callee(method_name.to_string(), callee_joules);
247                    }
248                }
249            }
250
251            ExprKind::Field { expr: inner, .. } => {
252                self.estimate_expr(inner, ctx);
253                // Field access is typically an offset calculation + load
254                ctx.add_cost(self.instruction_cost(InstructionType::IntAdd));
255                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1));
256            }
257
258            ExprKind::Index { expr: array, index } => {
259                self.estimate_expr(array, ctx);
260                self.estimate_expr(index, ctx);
261                // Index calculation + bounds check + load
262                ctx.add_cost(self.instruction_cost(InstructionType::IntMul));
263                ctx.add_cost(self.instruction_cost(InstructionType::IntAdd));
264                ctx.add_cost(self.instruction_cost(InstructionType::BranchNotTaken)); // bounds check
265                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1));
266            }
267
268            ExprKind::Tuple { elements } => {
269                for elem in elements {
270                    self.estimate_expr(elem, ctx);
271                }
272            }
273
274            ExprKind::Array { elements } => {
275                for elem in elements {
276                    self.estimate_expr(elem, ctx);
277                }
278                // Array allocation/storage
279                ctx.add_cost(
280                    self.instruction_cost(InstructionType::MemLoadL1) * elements.len() as f64,
281                );
282            }
283
284            ExprKind::ArrayRepeat { element, count } => {
285                self.estimate_expr(element, ctx);
286                // Repeated storage
287                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1) * *count as f64);
288            }
289
290            ExprKind::Struct { fields, .. } => {
291                for field in fields {
292                    self.estimate_expr(&field.value, ctx);
293                }
294                // Struct construction: store each field
295                ctx.add_cost(
296                    self.instruction_cost(InstructionType::MemLoadL1) * fields.len() as f64,
297                );
298            }
299
300            ExprKind::Block(block) => {
301                self.estimate_block(block, ctx);
302            }
303
304            ExprKind::If {
305                condition,
306                then_block,
307                else_block,
308            } => {
309                self.estimate_expr(condition, ctx);
310
311                // We don't know which branch will be taken at compile time,
312                // so estimate both and take the average
313                let mut then_ctx = ctx.clone();
314                self.estimate_block(then_block, &mut then_ctx);
315
316                let mut else_ctx = ctx.clone();
317                if let Some(else_expr) = else_block {
318                    self.estimate_expr(else_expr, &mut else_ctx);
319                }
320
321                // Average of both branches + branch cost
322                let then_cost = then_ctx.total_cost_pj - ctx.total_cost_pj;
323                let else_cost = else_ctx.total_cost_pj - ctx.total_cost_pj;
324                let avg_cost = f64::midpoint(then_cost, else_cost);
325
326                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken));
327                ctx.add_cost(avg_cost);
328
329                // Reduce confidence due to branching uncertainty
330                ctx.reduce_confidence(0.9);
331            }
332
333            ExprKind::Match {
334                expr: scrutinee,
335                arms,
336            } => {
337                self.estimate_expr(scrutinee, ctx);
338
339                // Estimate average cost across all arms
340                let mut total_arm_cost = 0.0;
341                for arm in arms {
342                    let mut arm_ctx = ctx.clone();
343                    self.estimate_expr(&arm.body, &mut arm_ctx);
344                    total_arm_cost += arm_ctx.total_cost_pj - ctx.total_cost_pj;
345                }
346
347                let avg_arm_cost = if !arms.is_empty() {
348                    total_arm_cost / arms.len() as f64
349                } else {
350                    0.0
351                };
352
353                // Branch cost per arm + average arm cost
354                ctx.add_cost(
355                    self.instruction_cost(InstructionType::BranchTaken) * arms.len() as f64,
356                );
357                ctx.add_cost(avg_arm_cost);
358
359                // Reduce confidence due to match uncertainty
360                ctx.reduce_confidence(0.85);
361            }
362
363            ExprKind::Loop { body } => {
364                // Track loop nesting
365                ctx.enter_loop();
366                let iterations = self.effective_loop_iterations() as f64;
367
368                // Unbounded loop - use default iterations with low confidence
369                let mut body_ctx = ctx.clone();
370                self.estimate_block(body, &mut body_ctx);
371                let body_cost = body_ctx.total_cost_pj - ctx.total_cost_pj;
372
373                ctx.add_cost(body_cost * iterations);
374                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken) * iterations);
375
376                // Very low confidence for unbounded loops, lower for nested
377                ctx.reduce_confidence(0.3 / ctx.loop_depth() as f64);
378                ctx.exit_loop();
379            }
380
381            ExprKind::While { condition, body } => {
382                ctx.enter_loop();
383                let iterations = self.effective_loop_iterations() as f64;
384
385                // Estimate condition + body cost per iteration
386                let mut cond_ctx = ctx.clone();
387                self.estimate_expr(condition, &mut cond_ctx);
388                let cond_cost = cond_ctx.total_cost_pj - ctx.total_cost_pj;
389
390                let mut body_ctx = ctx.clone();
391                self.estimate_block(body, &mut body_ctx);
392                let body_cost = body_ctx.total_cost_pj - ctx.total_cost_pj;
393
394                let iteration_cost =
395                    cond_cost + body_cost + self.instruction_cost(InstructionType::BranchTaken);
396                ctx.add_cost(iteration_cost * iterations);
397
398                // Low confidence for while loops, lower for nested
399                ctx.reduce_confidence(0.4 / ctx.loop_depth() as f64);
400                ctx.exit_loop();
401            }
402
403            ExprKind::For { iter, body, .. } => {
404                ctx.enter_loop();
405                let iterations = self.effective_loop_iterations() as f64;
406
407                self.estimate_expr(iter, ctx);
408
409                // Estimate body cost per iteration
410                let mut body_ctx = ctx.clone();
411                self.estimate_block(body, &mut body_ctx);
412                let body_cost = body_ctx.total_cost_pj - ctx.total_cost_pj;
413
414                // For loops often have known bounds, so higher confidence than while
415                ctx.add_cost(body_cost * iterations);
416                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken) * iterations);
417                ctx.add_cost(self.instruction_cost(InstructionType::IntAdd) * iterations); // increment
418
419                // Medium confidence for for loops, lower for nested
420                ctx.reduce_confidence(0.6 / ctx.loop_depth() as f64);
421                ctx.exit_loop();
422            }
423
424            ExprKind::Return { value } => {
425                if let Some(val) = value {
426                    self.estimate_expr(val, ctx);
427                }
428                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken));
429            }
430
431            ExprKind::Break { value } => {
432                if let Some(val) = value {
433                    self.estimate_expr(val, ctx);
434                }
435                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken));
436            }
437
438            ExprKind::Continue => {
439                ctx.add_cost(self.instruction_cost(InstructionType::BranchTaken));
440            }
441
442            ExprKind::Assign { target, value } => {
443                self.estimate_expr(target, ctx);
444                self.estimate_expr(value, ctx);
445                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1)); // store
446            }
447
448            ExprKind::Ref { expr: inner, .. } => {
449                self.estimate_expr(inner, ctx);
450                ctx.add_cost(self.instruction_cost(InstructionType::IntAdd)); // address calculation
451            }
452
453            ExprKind::Deref { expr: inner } => {
454                self.estimate_expr(inner, ctx);
455                // Dereference is a memory load
456                ctx.add_cost(self.instruction_cost(InstructionType::MemLoadL1));
457            }
458
459            ExprKind::Cast { expr: inner, .. } => {
460                self.estimate_expr(inner, ctx);
461                // Most casts are free or very cheap
462                ctx.add_cost(0.05);
463            }
464
465            // ===== Concurrency Expressions =====
466            // These have higher overhead due to task management and synchronization
467            ExprKind::Spawn { expr: inner } => {
468                self.estimate_expr(inner, ctx);
469                // Task spawn overhead: allocation, queue insertion, scheduling
470                ctx.add_cost(50.0); // ~50 μJ for task creation
471            }
472
473            ExprKind::SpawnWithDeadline { deadline, body } => {
474                self.estimate_expr(deadline, ctx);
475                self.estimate_block(body, ctx);
476                // Extra overhead for deadline tracking
477                ctx.add_cost(75.0);
478            }
479
480            ExprKind::TaskGroup { body } | ExprKind::TaskGroupAll { body } => {
481                self.estimate_block(body, ctx);
482                // Task group management overhead
483                ctx.add_cost(25.0);
484            }
485
486            ExprKind::Channel { capacity, .. } => {
487                self.estimate_expr(capacity, ctx);
488                // Channel allocation and setup
489                ctx.add_cost(30.0);
490            }
491
492            ExprKind::Select { arms } | ExprKind::SelectBiased { arms } => {
493                for arm in arms {
494                    match &arm.operation {
495                        HirSelectOperation::Recv { channel } => {
496                            self.estimate_expr(channel, ctx);
497                        }
498                        HirSelectOperation::Send { channel, value } => {
499                            self.estimate_expr(channel, ctx);
500                            self.estimate_expr(value, ctx);
501                        }
502                        HirSelectOperation::Timeout { duration } => {
503                            self.estimate_expr(duration, ctx);
504                        }
505                    }
506                    self.estimate_expr(&arm.body, ctx);
507                }
508                // Select operation overhead
509                ctx.add_cost(20.0);
510            }
511
512            ExprKind::Gpu { body } => {
513                self.estimate_block(body, ctx);
514                // GPU transfer and execution overhead
515                ctx.add_cost(100.0); // GPU operations are more energy-intensive
516            }
517
518            ExprKind::Distributed { body } => {
519                self.estimate_block(body, ctx);
520                // Network and coordination overhead
521                ctx.add_cost(200.0);
522            }
523
524            ExprKind::SpawnRemote { node, expr: inner } => {
525                self.estimate_expr(node, ctx);
526                self.estimate_expr(inner, ctx);
527                // Remote spawn has significant network overhead
528                ctx.add_cost(500.0);
529            }
530
531            ExprKind::Cancelled => {
532                // Simple flag check
533                ctx.add_cost(0.5);
534            }
535
536            ExprKind::Try { expr: inner } => {
537                self.estimate_expr(inner, ctx);
538                ctx.add_cost(1.0);
539            }
540
541            ExprKind::Path { .. } => {
542                // Path expressions (like enum variants) are essentially constant
543                ctx.add_cost(0.01);
544            }
545
546            ExprKind::Closure { body, captures, .. } => {
547                // Closure creation: cost of capturing variables + body analysis
548                ctx.add_cost(captures.len() as f64 * 1.0);
549                self.estimate_expr(body, ctx);
550            }
551
552            ExprKind::EnergyBudgetBlock { body, .. } => {
553                self.estimate_block(body, ctx);
554            }
555
556            ExprKind::ThermalAdapt { arms } => {
557                // Estimate cost of each arm; only one runs at runtime
558                for arm in arms {
559                    self.estimate_expr(&arm.body, ctx);
560                }
561                // Thermal state check overhead
562                ctx.add_cost(5.0);
563            }
564        }
565    }

567    /// Get the energy cost for a binary operation based on operand type
568    fn binary_op_cost(&self, op: BinOp, ty: &Ty) -> f64 {
569        let instr = match (op, ty) {
570            // Integer operations
571            (BinOp::Add, Ty::Int(_) | Ty::Uint(_)) => InstructionType::IntAdd,
572            (BinOp::Sub, Ty::Int(_) | Ty::Uint(_)) => InstructionType::IntSub,
573            (BinOp::Mul, Ty::Int(_) | Ty::Uint(_)) => InstructionType::IntMul,
574            (BinOp::Div, Ty::Int(_) | Ty::Uint(_)) => InstructionType::IntDiv,
575            (BinOp::Rem, Ty::Int(_) | Ty::Uint(_)) => InstructionType::IntRem,
576
577            // Float operations
578            (BinOp::Add, Ty::Float(_)) => InstructionType::FloatAdd,
579            (BinOp::Sub, Ty::Float(_)) => InstructionType::FloatSub,
580            (BinOp::Mul, Ty::Float(_)) => InstructionType::FloatMul,
581            (BinOp::Div, Ty::Float(_)) => InstructionType::FloatDiv,
582
583            // Bitwise operations
584            (BinOp::BitAnd, _) => InstructionType::BitAnd,
585            (BinOp::BitOr, _) => InstructionType::BitOr,
586            (BinOp::BitXor, _) => InstructionType::BitXor,
587            (BinOp::Shl, _) => InstructionType::BitShl,
588            (BinOp::Shr, _) => InstructionType::BitShr,
589
590            // Logical operations (branches)
591            (BinOp::And | BinOp::Or, _) => InstructionType::BranchTaken,
592
593            // Comparisons
594            (BinOp::Eq | BinOp::Ne | BinOp::Lt | BinOp::Le | BinOp::Gt | BinOp::Ge, _) => {
595                InstructionType::IntSub // Compare is typically subtract + flags
596            }
597
598            // Default
599            _ => InstructionType::IntAdd,
600        };
601
602        self.instruction_cost(instr)
603    }

605    /// Get the energy cost for a unary operation
606    fn unary_op_cost(&self, op: UnOp, ty: &Ty) -> f64 {
607        let instr = match (op, ty) {
608            (UnOp::Neg, Ty::Float(_)) => InstructionType::FloatSub, // 0 - x
609            (UnOp::Neg, _) => InstructionType::IntSub,              // 0 - x
610            (UnOp::Not, Ty::Bool) => InstructionType::BitXor,       // xor with 1
611            (UnOp::Not, _) => InstructionType::BitXor,              // bitwise not
612        };
613
614        self.instruction_cost(instr)
615    }

617    /// Extract the function name from a Call expression's callee
618    fn extract_callee_name(callee_expr: &Expr) -> Option<String> {
619        match &callee_expr.kind {
620            ExprKind::Var { name, .. } => Some(name.as_str()),
621            ExprKind::Path { segments } => {
622                // Use last segment as the function name
623                segments.last().map(|seg| seg.as_str())
624            }
625            _ => None,
626        }
627    }

629    /// Estimate a function's energy with transitive callee costs.
630    ///
631    /// Like `estimate_function`, but includes the estimated energy of called
632    /// functions. Returns (joules, confidence, top_callees) where top_callees
633    /// lists each callee and its energy contribution.
634    pub fn estimate_function_transitive(
635        &self,
636        func: &Function,
637        callee_costs: &std::collections::HashMap<String, f64>,
638    ) -> (f64, f64, Vec<(String, f64)>) {
639        let mut ctx = EstimationContext::with_callee_costs(callee_costs.clone());
640        self.estimate_block(&func.body, &mut ctx);
641
642        let joules = ctx.total_cost_pj / 1e12;
643
644        // Deduplicate and sort callee contributions (sum by name, descending)
645        let mut callee_sums: std::collections::HashMap<String, f64> =
646            std::collections::HashMap::new();
647        for (name, cost) in &ctx.top_callees {
648            *callee_sums.entry(name.clone()).or_insert(0.0) += cost;
649        }
650        let mut sorted_callees: Vec<(String, f64)> = callee_sums.into_iter().collect();
651        sorted_callees.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
652
653        (joules, ctx.confidence, sorted_callees)
654    }

656    /// Build an energy map from function name to estimated joules.
657    ///
658    /// This is the seed for transitive analysis — each function's local-only
659    /// estimate, keyed by name.
660    pub fn build_energy_map(functions: &[Function]) -> std::collections::HashMap<String, f64> {
661        let mut map = std::collections::HashMap::new();
662        for func in functions {
663            if let Some(cost) = func.energy.estimated_cost {
664                map.insert(func.name.as_str(), cost);
665            }
666        }

670    /// Check if a function exceeds its declared energy budget.
671    ///
672    /// Returns Some(EnergyViolation) if the estimated energy exceeds the budget,
673    /// None otherwise.
674    pub fn check_budget(&self, func: &Function) -> Option<EnergyViolation> {
675        let budget = func.energy.budget.as_ref()?;
676
677        if !budget.should_check() {
678            return None;
679        }
680
681        let (estimated_joules, confidence) = self.estimate_function(func);
682
683        // Check joule budget
684        if let Some(max_joules) = budget.max_joules {
685            if estimated_joules > max_joules {
686                return Some(EnergyViolation::joules(
687                    max_joules,
688                    estimated_joules,
689                    budget.span,
690                    func.span,
691                    confidence,
692                ));
693            }
694        }
695
696        // Check power budget (watts)
697        if let Some(max_watts) = budget.max_watts {
698            let (estimated_watts, power_confidence) = self.estimate_function_power(func);
699            if estimated_watts > max_watts {
700                return Some(EnergyViolation::watts(
701                    max_watts,
702                    estimated_watts,
703                    budget.span,
704                    func.span,
705                    power_confidence,
706                ));
707            }
708        }
709
710        // Check temperature budget (Celsius delta)
711        if let Some(max_temp_delta) = budget.max_temp_delta {
712            let (estimated_temp, temp_confidence) = self.estimate_function_temp_delta(func);
713            if estimated_temp > max_temp_delta {
714                return Some(EnergyViolation::temp_delta(
715                    max_temp_delta,
716                    estimated_temp,
717                    budget.span,
718                    func.span,
719                    temp_confidence,
720                ));
721            }
722        }
723
724        None
725    }

731        } else {

736    /// Create a diagnostic for an energy violation
737    pub fn violation_to_diagnostic(
738        &self,
739        violation: &EnergyViolation,
740        func_name: &str,
741    ) -> Diagnostic {
742        let message = format!("function '{}' {}", func_name, violation.message);
743
744        let mut diag = if violation.confidence > 0.8 {
745            Diagnostic::error(&message)
746        } else {
747            Diagnostic::warning(&message)
748        };
749
750        let unit = match violation.kind {
751            joule_hir::EnergyViolationKind::JoulesExceeded => "J",
752            joule_hir::EnergyViolationKind::WattsExceeded => "W",
753            joule_hir::EnergyViolationKind::TempDeltaExceeded => "°C",
754        };
755
756        let est_str = Self::format_with_unit(violation.estimated, unit);
757        let bud_str = Self::format_with_unit(violation.budget, unit);
758
759        diag = diag.with_label(Label::primary(
760            violation.code_span,
761            format!("estimated at {est_str}"),
762        ));
763
764        diag = diag.with_label(Label::secondary(
765            violation.budget_span,
766            format!("budget of {bud_str} declared here"),
767        ));
768
769        if violation.confidence < 1.0 {
770            diag = diag.with_note(format!(
771                "estimate confidence: {:.0}% (due to loops with unknown bounds)",
772                violation.confidence * 100.0
773            ));
774        }
775
776        diag = diag.with_code(format!(
777            "E{}",
778            match violation.kind {
779                joule_hir::EnergyViolationKind::JoulesExceeded => "0601",
780                joule_hir::EnergyViolationKind::WattsExceeded => "0602",
781                joule_hir::EnergyViolationKind::TempDeltaExceeded => "0603",
782            }
783        ));
784
785        diag
786    }

788    /// Create a diagnostic for an energy violation that includes callee chain details
789    pub fn violation_to_diagnostic_transitive(
790        &self,
791        violation: &EnergyViolation,
792        func_name: &str,
793        callee_chain: &[(String, f64)],
794    ) -> Diagnostic {
795        let mut diag = self.violation_to_diagnostic(violation, func_name);
796
797        // Add callee chain notes showing where the energy went
798        if !callee_chain.is_empty() {
799            let mut note = "significant callee costs (transitive):".to_string();
800            for (callee_name, callee_cost) in callee_chain.iter().take(5) {
801                let cost_str = EnergyViolation::format_joules(*callee_cost);
802                note.push_str(&format!("\n  {}() ~{}", callee_name, cost_str));
803            }
804            if callee_chain.len() > 5 {
805                note.push_str(&format!(
806                    "\n  ... and {} more callees",
807                    callee_chain.len() - 5
808                ));
809            }
810            diag = diag.with_note(note);
811        }
812
813        diag
814    }

816    /// Check if a function exceeds its budget with transitive callee costs.
817    ///
818    /// Returns Some((violation, callee_chain)) if the transitive estimate
819    /// exceeds the budget.
820    pub fn check_budget_transitive(
821        &self,
822        func: &Function,
823        callee_costs: &std::collections::HashMap<String, f64>,
824    ) -> Option<(EnergyViolation, Vec<(String, f64)>)> {
825        let budget = func.energy.budget.as_ref()?;
826
827        if !budget.should_check() {
828            return None;
829        }
830
831        let (estimated_joules, confidence, callee_chain) =
832            self.estimate_function_transitive(func, callee_costs);
833
834        if let Some(max_joules) = budget.max_joules {
835            if estimated_joules > max_joules {
836                let violation = EnergyViolation::joules(
837                    max_joules,
838                    estimated_joules,
839                    budget.span,
840                    func.span,
841                    confidence,
842                );
843                return Some((violation, callee_chain));
844            }
845        }
846
847        None
848    }

850    /// Estimate energy for a function and update its metadata
851    pub fn analyze_function(&self, func: &mut Function) -> Option<EnergyViolation> {
852        let (estimated_joules, confidence) = self.estimate_function(func);
853        let (estimated_watts, _) = self.estimate_function_power(func);
854        let (estimated_temp, _) = self.estimate_function_temp_delta(func);
855
856        // Update the function's energy metadata with our estimates
857        func.energy.estimated_cost = Some(estimated_joules);
858        func.energy.estimated_power = Some(estimated_watts);
859        func.energy.estimated_temp_delta = Some(estimated_temp);
860        func.energy.confidence = confidence;
861
862        // Check for violations
863        self.check_budget(func)
864    }
865}

873/// Context for energy estimation, tracking accumulated costs
874#[derive(Debug, Clone)]
875struct EstimationContext {
876    /// Total accumulated energy cost in picojoules
877    total_cost_pj: f64,
878    /// Confidence level (0.0 to 1.0)
879    confidence: f64,
880    /// Current loop nesting depth
881    loop_depth: u32,
882    /// Callee energy costs in joules (function name → joules), for transitive analysis
883    callee_costs: Option<std::collections::HashMap<String, f64>>,
884    /// Top callee contributors: (name, cost_joules) for diagnostics
885    top_callees: Vec<(String, f64)>,
886}
887
888impl EstimationContext {
889    fn new() -> Self {
890        Self {
891            total_cost_pj: 0.0,
892            confidence: 1.0,
893            loop_depth: 0,
894            callee_costs: None,
895            top_callees: Vec::new(),
896        }
897    }
898
899    fn with_callee_costs(callee_costs: std::collections::HashMap<String, f64>) -> Self {
900        Self {
901            total_cost_pj: 0.0,
902            confidence: 1.0,
903            loop_depth: 0,
904            callee_costs: Some(callee_costs),
905            top_callees: Vec::new(),
906        }
907    }
908
909    fn record_callee(&mut self, name: String, cost_joules: f64) {
910        self.top_callees.push((name, cost_joules));
911    }
912
913    fn add_cost(&mut self, cost_pj: f64) {
914        self.total_cost_pj += cost_pj;
915    }
916
917    fn reduce_confidence(&mut self, factor: f64) {
918        self.confidence *= factor;
919        self.confidence = self.confidence.max(0.1); // Minimum 10% confidence
920    }
921
922    fn enter_loop(&mut self) {
923        self.loop_depth += 1;
924    }
925
926    fn exit_loop(&mut self) {
927        self.loop_depth = self.loop_depth.saturating_sub(1);
928    }
929
930    fn loop_depth(&self) -> u32 {
931        self.loop_depth.max(1) // At least 1 to avoid division by zero
932    }
933}
