1//! Monomorphization for Joule
2//!
3//! Converts generic code to specialized versions for each concrete type instantiation.
4//! This module handles:
5//! - Collecting all generic instantiations from the program
6//! - Generating specialized versions of generic functions and types
7//! - Name mangling for different instantiations
8//! - Integration with MIR generation
9
10use std::collections::{HashMap, HashSet, VecDeque};
11use std::fmt;

13/// A monomorphization request (generic item + concrete type arguments)
14#[derive(Debug, Clone, PartialEq, Eq, Hash)]
15pub struct MonoItem {
16    /// The generic item being instantiated
17    pub item: ItemId,
18    /// The concrete type arguments
19    pub args: Vec<MonoType>,
20}
21
22impl MonoItem {
23    /// Create a new monomorphization item
24    pub fn new(item: ItemId, args: Vec<MonoType>) -> Self {
25        Self { item, args }
26    }
27
28    /// Check if this item has no type arguments (non-generic)
29    pub fn is_concrete(&self) -> bool {
30        self.args.is_empty()
31    }
32
33    /// Get the mangled name for this instantiation
34    pub fn mangled_name(&self, base_name: &str) -> String {
35        mangle_name(base_name, &self.args)
36    }
37}

43impl ItemId {
44    pub fn new(id: u32) -> Self {

49/// A monomorphized type (fully concrete)
50#[derive(Debug, Clone, PartialEq, Eq, Hash)]
51pub enum MonoType {
52    /// A primitive type
53    Primitive(PrimitiveType),
54    /// A struct or enum with concrete type arguments
55    Adt { item: ItemId, args: Vec<MonoType> },
56    /// A tuple
57    Tuple(Vec<MonoType>),
58    /// A reference
59    Ref { mutable: bool, inner: Box<MonoType> },
60    /// An array
61    Array { elem: Box<MonoType>, size: usize },
62    /// A slice
63    Slice(Box<MonoType>),
64    /// A function pointer
65    FnPtr {
66        params: Vec<MonoType>,
67        ret: Box<MonoType>,
68    },
69    /// The never type
70    Never,
71    /// Unit type
72    Unit,
73}
74
75impl MonoType {
76    /// Get the size of this type in bytes (approximation for layout)
77    pub fn size_bytes(&self) -> usize {
78        match self {
79            MonoType::Primitive(p) => p.size_bytes(),
80            MonoType::Adt { args, .. } => {
81                // Conservative estimate - sum of arg sizes
82                args.iter().map(|a| a.size_bytes()).sum::<usize>().max(1)
83            }
84            MonoType::Tuple(elems) => elems.iter().map(|e| e.size_bytes()).sum(),
85            MonoType::Ref { .. } => 8, // Pointer size
86            MonoType::Array { elem, size } => elem.size_bytes() * size,
87            MonoType::Slice(_) => 16, // Fat pointer (ptr + len)
88            MonoType::FnPtr { .. } => 8,
89            MonoType::Never => 0,
90            MonoType::Unit => 0,
91        }
92    }
93
94    /// Get the alignment of this type in bytes
95    pub fn alignment(&self) -> usize {
96        match self {
97            MonoType::Primitive(p) => p.alignment(),
98            MonoType::Adt { args, .. } => args.iter().map(|a| a.alignment()).max().unwrap_or(1),
99            MonoType::Tuple(elems) => elems.iter().map(|e| e.alignment()).max().unwrap_or(1),
100            MonoType::Ref { .. } => 8,
101            MonoType::Array { elem, .. } => elem.alignment(),
102            MonoType::Slice(_) => 8,
103            MonoType::FnPtr { .. } => 8,
104            MonoType::Never => 1,
105            MonoType::Unit => 1,
106        }
107    }
108
109    /// Check if this type is zero-sized
110    pub fn is_zst(&self) -> bool {
111        self.size_bytes() == 0
112    }
113
114    /// Check if this type is a primitive
115    pub fn is_primitive(&self) -> bool {
116        matches!(self, MonoType::Primitive(_))
117    }
118}
119
120impl fmt::Display for MonoType {
121    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
122        write!(f, "{}", mangle_type(self))
123    }
124}

145impl PrimitiveType {
146    /// Get the size in bytes

148        match self {
149            PrimitiveType::Bool | PrimitiveType::I8 | PrimitiveType::U8 => 1,
150            PrimitiveType::I16 | PrimitiveType::U16 => 2,
151            PrimitiveType::I32 | PrimitiveType::U32 | PrimitiveType::F32 | PrimitiveType::Char => 4,
152            PrimitiveType::I64 | PrimitiveType::U64 | PrimitiveType::F64 => 8,
153            PrimitiveType::I128 | PrimitiveType::U128 => 16,
154        }

159        self.size_bytes().min(8) // Max alignment is typically 8 bytes
160    }

164        matches!(

166            PrimitiveType::I8

169                | PrimitiveType::I64
170                | PrimitiveType::I128
171        )
172    }

178            PrimitiveType::U8

180                | PrimitiveType::U32
181                | PrimitiveType::U64
182                | PrimitiveType::U128
183        )
184    }

197/// The monomorphization collector
198#[derive(Debug)]
199pub struct MonoCollector {
200    /// Items that need to be monomorphized
201    worklist: VecDeque<MonoItem>,
202    /// Items that have been processed
203    processed: HashSet<MonoItem>,
204    /// Map from generic item to its monomorphized versions
205    instances: HashMap<ItemId, Vec<MonoItem>>,
206    /// Item names for mangling
207    item_names: HashMap<ItemId, String>,
208}
209
210impl MonoCollector {
211    /// Create a new monomorphization collector
212    pub fn new() -> Self {
213        Self {
214            worklist: VecDeque::new(),
215            processed: HashSet::new(),
216            instances: HashMap::new(),
217            item_names: HashMap::new(),
218        }
219    }
220
221    /// Register an item name for mangling
222    pub fn register_item_name(&mut self, item: ItemId, name: String) {
223        self.item_names.insert(item, name);
224    }
225
226    /// Get an item's name
227    pub fn get_item_name(&self, item: ItemId) -> Option<&str> {
228        self.item_names.get(&item).map(String::as_str)
229    }
230
231    /// Add an item to the monomorphization worklist
232    pub fn add_item(&mut self, item: MonoItem) {
233        if !self.processed.contains(&item) && !self.worklist.contains(&item) {
234            self.worklist.push_back(item.clone());
235            self.instances.entry(item.item).or_default().push(item);
236        }
237    }
238
239    /// Add a root item (entry point or public API)
240    pub fn add_root(&mut self, item: MonoItem) {
241        self.add_item(item);
242    }
243
244    /// Get the next item to process
245    pub fn next(&mut self) -> Option<MonoItem> {
246        while let Some(item) = self.worklist.pop_front() {
247            if !self.processed.contains(&item) {
248                self.processed.insert(item.clone());
249                return Some(item);
250            }
251        }
252        None
253    }
254
255    /// Check if collection is complete
256    pub fn is_done(&self) -> bool {
257        self.worklist.is_empty()
258    }
259
260    /// Get all instances of a generic item
261    pub fn get_instances(&self, item: ItemId) -> &[MonoItem] {
262        self.instances.get(&item).map_or(&[], Vec::as_slice)
263    }
264
265    /// Get all processed items
266    pub fn all_items(&self) -> impl Iterator<Item = &MonoItem> {
267        self.processed.iter()
268    }
269
270    /// Get the number of processed items
271    pub fn processed_count(&self) -> usize {
272        self.processed.len()
273    }
274
275    /// Get the mangled name for a mono item
276    pub fn mangled_name(&self, item: &MonoItem) -> String {
277        let base_name = self
278            .item_names
279            .get(&item.item)
280            .map(String::as_str)
281            .unwrap_or("unknown");
282        mangle_name(base_name, &item.args)
283    }
284
285    /// Clear all collected items (useful for incremental compilation)
286    pub fn clear(&mut self) {
287        self.worklist.clear();
288        self.processed.clear();
289        self.instances.clear();
290    }
291}
292
293impl Default for MonoCollector {
294    fn default() -> Self {
295        Self::new()
296    }
297}

299/// A generic type (may contain type parameters)
300#[derive(Debug, Clone, PartialEq, Eq, Hash)]
301pub enum GenericType {
302    /// A type parameter
303    Param(u32),
304    /// A primitive
305    Primitive(PrimitiveType),
306    /// An ADT
307    Adt {
308        item: ItemId,
309        args: Vec<GenericType>,
310    },
311    /// A tuple
312    Tuple(Vec<GenericType>),
313    /// A reference
314    Ref {
315        mutable: bool,
316        inner: Box<GenericType>,
317    },
318    /// An array
319    Array { elem: Box<GenericType>, size: usize },
320    /// A slice
321    Slice(Box<GenericType>),
322    /// A function pointer
323    FnPtr {
324        params: Vec<GenericType>,
325        ret: Box<GenericType>,
326    },
327    /// Never type
328    Never,
329    /// Unit type
330    Unit,
331}
332
333impl GenericType {
334    /// Check if this type contains any type parameters
335    pub fn has_params(&self) -> bool {
336        match self {
337            GenericType::Param(_) => true,
338            GenericType::Primitive(_) | GenericType::Never | GenericType::Unit => false,
339            GenericType::Adt { args, .. } => args.iter().any(|a| a.has_params()),
340            GenericType::Tuple(elems) => elems.iter().any(|e| e.has_params()),
341            GenericType::Ref { inner, .. } => inner.has_params(),
342            GenericType::Array { elem, .. } => elem.has_params(),
343            GenericType::Slice(inner) => inner.has_params(),
344            GenericType::FnPtr { params, ret } => {
345                params.iter().any(|p| p.has_params()) || ret.has_params()
346            }
347        }
348    }
349
350    /// Get all type parameter indices used in this type
351    pub fn param_indices(&self) -> Vec<u32> {
352        let mut indices = Vec::new();
353        self.collect_param_indices(&mut indices);
354        indices.sort_unstable();
355        indices.dedup();
356        indices
357    }
358
359    fn collect_param_indices(&self, indices: &mut Vec<u32>) {
360        match self {
361            GenericType::Param(idx) => indices.push(*idx),
362            GenericType::Primitive(_) | GenericType::Never | GenericType::Unit => {}
363            GenericType::Adt { args, .. } => {
364                for arg in args {
365                    arg.collect_param_indices(indices);
366                }
367            }
368            GenericType::Tuple(elems) => {
369                for elem in elems {
370                    elem.collect_param_indices(indices);
371                }
372            }
373            GenericType::Ref { inner, .. }
374            | GenericType::Array { elem: inner, .. }
375            | GenericType::Slice(inner) => {
376                inner.collect_param_indices(indices);
377            }
378            GenericType::FnPtr { params, ret } => {
379                for p in params {
380                    p.collect_param_indices(indices);
381                }
382                ret.collect_param_indices(indices);
383            }
384        }
385    }
386}

388/// Substitution from type parameters to concrete types
389#[derive(Debug, Clone, Default)]
390pub struct MonoSubst {
391    types: HashMap<u32, MonoType>,
392}
393
394impl MonoSubst {
395    /// Create an empty substitution
396    pub fn empty() -> Self {
397        Self::default()
398    }
399
400    /// Create from a list of types (indexed 0, 1, 2, ...)
401    pub fn from_types(types: Vec<MonoType>) -> Self {
402        let mut subst = Self::empty();
403        for (i, ty) in types.into_iter().enumerate() {
404            subst.types.insert(i as u32, ty);
405        }
406        subst
407    }
408
409    /// Create from type arguments with explicit indices
410    pub fn from_indexed_types(indexed_types: impl IntoIterator<Item = (u32, MonoType)>) -> Self {
411        let mut subst = Self::empty();
412        for (idx, ty) in indexed_types {
413            subst.types.insert(idx, ty);
414        }
415        subst
416    }
417
418    /// Get a type by parameter index
419    pub fn get(&self, idx: u32) -> Option<&MonoType> {
420        self.types.get(&idx)
421    }
422
423    /// Insert a type mapping
424    pub fn insert(&mut self, idx: u32, ty: MonoType) {
425        self.types.insert(idx, ty);
426    }
427
428    /// Check if this substitution is empty
429    pub fn is_empty(&self) -> bool {
430        self.types.is_empty()
431    }
432
433    /// Get the number of type mappings
434    pub fn len(&self) -> usize {
435        self.types.len()
436    }
437
438    /// Iterate over all mappings
439    pub fn iter(&self) -> impl Iterator<Item = (&u32, &MonoType)> {
440        self.types.iter()
441    }
442}

444/// Apply a substitution to get a monomorphized type
445pub fn monomorphize_type(ty: &GenericType, subst: &MonoSubst) -> MonoType {
446    match ty {
447        GenericType::Param(idx) => subst.get(*idx).cloned().unwrap_or(MonoType::Unit),
448        GenericType::Primitive(p) => MonoType::Primitive(*p),
449        GenericType::Adt { item, args } => MonoType::Adt {
450            item: *item,
451            args: args.iter().map(|a| monomorphize_type(a, subst)).collect(),
452        },
453        GenericType::Tuple(elems) => {
454            MonoType::Tuple(elems.iter().map(|e| monomorphize_type(e, subst)).collect())
455        }
456        GenericType::Ref { mutable, inner } => MonoType::Ref {
457            mutable: *mutable,
458            inner: Box::new(monomorphize_type(inner, subst)),
459        },
460        GenericType::Array { elem, size } => MonoType::Array {
461            elem: Box::new(monomorphize_type(elem, subst)),
462            size: *size,
463        },
464        GenericType::Slice(inner) => MonoType::Slice(Box::new(monomorphize_type(inner, subst))),
465        GenericType::FnPtr { params, ret } => MonoType::FnPtr {
466            params: params.iter().map(|p| monomorphize_type(p, subst)).collect(),
467            ret: Box::new(monomorphize_type(ret, subst)),
468        },
469        GenericType::Never => MonoType::Never,
470        GenericType::Unit => MonoType::Unit,
471    }
472}

474/// Generate a mangled name for a monomorphized item
475pub fn mangle_name(base_name: &str, args: &[MonoType]) -> String {
476    if args.is_empty() {
477        return base_name.to_string();
478    }
479
480    let mut name = base_name.to_string();
481    name.push_str("__");
482
483    for (i, arg) in args.iter().enumerate() {
484        if i > 0 {
485            name.push('_');
486        }
487        name.push_str(&mangle_type(arg));
488    }
489
490    name
491}

493/// Mangle a type for name generation
494fn mangle_type(ty: &MonoType) -> String {
495    match ty {
496        MonoType::Primitive(p) => match p {
497            PrimitiveType::Bool => "bool".to_string(),
498            PrimitiveType::I8 => "i8".to_string(),
499            PrimitiveType::I16 => "i16".to_string(),
500            PrimitiveType::I32 => "i32".to_string(),
501            PrimitiveType::I64 => "i64".to_string(),
502            PrimitiveType::I128 => "i128".to_string(),
503            PrimitiveType::U8 => "u8".to_string(),
504            PrimitiveType::U16 => "u16".to_string(),
505            PrimitiveType::U32 => "u32".to_string(),
506            PrimitiveType::U64 => "u64".to_string(),
507            PrimitiveType::U128 => "u128".to_string(),
508            PrimitiveType::F32 => "f32".to_string(),
509            PrimitiveType::F64 => "f64".to_string(),
510            PrimitiveType::Char => "char".to_string(),
511        },
512        MonoType::Adt { item, args } => {
513            let mut s = format!("adt{}", item.0);
514            if !args.is_empty() {
515                s.push('<');
516                for (i, arg) in args.iter().enumerate() {
517                    if i > 0 {
518                        s.push(',');
519                    }
520                    s.push_str(&mangle_type(arg));
521                }
522                s.push('>');
523            }
524            s
525        }
526        MonoType::Tuple(elems) => {
527            if elems.is_empty() {
528                "unit".to_string()
529            } else {
530                let mut s = "tuple".to_string();
531                s.push('<');
532                for (i, elem) in elems.iter().enumerate() {
533                    if i > 0 {
534                        s.push(',');
535                    }
536                    s.push_str(&mangle_type(elem));
537                }
538                s.push('>');
539                s
540            }
541        }
542        MonoType::Ref { mutable, inner } => {
543            let prefix = if *mutable { "mutref" } else { "ref" };
544            format!("{}<{}>", prefix, mangle_type(inner))
545        }
546        MonoType::Array { elem, size } => {
547            format!("array<{},{}>", mangle_type(elem), size)
548        }
549        MonoType::Slice(inner) => {
550            format!("slice<{}>", mangle_type(inner))
551        }
552        MonoType::FnPtr { params, ret } => {
553            let mut s = "fn".to_string();
554            s.push('(');
555            for (i, param) in params.iter().enumerate() {
556                if i > 0 {
557                    s.push(',');
558                }
559                s.push_str(&mangle_type(param));
560            }
561            s.push_str(")->");
562            s.push_str(&mangle_type(ret));
563            s
564        }
565        MonoType::Never => "never".to_string(),
566        MonoType::Unit => "unit".to_string(),
567    }
568}

570/// Demangle a type name back to a MonoType (best effort)
571/// Returns None if demangling fails
572pub fn demangle_name(mangled: &str) -> Option<(String, Vec<MonoType>)> {
573    if let Some(pos) = mangled.find("__") {
574        let base_name = &mangled[..pos];
575        let type_part = &mangled[pos + 2..];
576
577        // Split on '_' (but not inside < >)
578        let mut args = Vec::new();
579        let mut current = String::new();
580        let mut depth = 0;
581
582        for c in type_part.chars() {
583            match c {
584                '<' => {
585                    depth += 1;
586                    current.push(c);
587                }
588                '>' => {
589                    depth -= 1;
590                    current.push(c);
591                }
592                '_' if depth == 0 => {
593                    if !current.is_empty() {
594                        if let Some(ty) = demangle_type(&current) {
595                            args.push(ty);
596                        }
597                        current.clear();
598                    }
599                }
600                _ => current.push(c),
601            }
602        }
603
604        if !current.is_empty() {
605            if let Some(ty) = demangle_type(&current) {
606                args.push(ty);
607            }
608        }
609
610        Some((base_name.to_string(), args))
611    } else {
612        Some((mangled.to_string(), Vec::new()))
613    }
614}

616fn demangle_type(s: &str) -> Option<MonoType> {
617    match s {
618        "bool" => Some(MonoType::Primitive(PrimitiveType::Bool)),
619        "i8" => Some(MonoType::Primitive(PrimitiveType::I8)),
620        "i16" => Some(MonoType::Primitive(PrimitiveType::I16)),
621        "i32" => Some(MonoType::Primitive(PrimitiveType::I32)),
622        "i64" => Some(MonoType::Primitive(PrimitiveType::I64)),
623        "i128" => Some(MonoType::Primitive(PrimitiveType::I128)),
624        "u8" => Some(MonoType::Primitive(PrimitiveType::U8)),
625        "u16" => Some(MonoType::Primitive(PrimitiveType::U16)),
626        "u32" => Some(MonoType::Primitive(PrimitiveType::U32)),
627        "u64" => Some(MonoType::Primitive(PrimitiveType::U64)),
628        "u128" => Some(MonoType::Primitive(PrimitiveType::U128)),
629        "f32" => Some(MonoType::Primitive(PrimitiveType::F32)),
630        "f64" => Some(MonoType::Primitive(PrimitiveType::F64)),
631        "char" => Some(MonoType::Primitive(PrimitiveType::Char)),
632        "unit" => Some(MonoType::Unit),
633        "never" => Some(MonoType::Never),
634        _ => None, // Complex types not supported in demangling
635    }

638/// Statistics about monomorphization
639#[derive(Debug, Default)]
640pub struct MonoStats {
641    /// Number of generic functions
642    pub generic_functions: usize,
643    /// Number of monomorphized instances
644    pub mono_instances: usize,
645    /// Number of unique type combinations
646    pub unique_type_combos: usize,
647    /// Total code size increase factor (estimate)
648    pub code_size_factor: f64,
649}
650
651impl MonoStats {
652    /// Calculate statistics from a collector
653    pub fn from_collector(collector: &MonoCollector) -> Self {
654        let generic_functions = collector.instances.len();
655        let mono_instances = collector.processed.len();
656        let unique_type_combos = collector.instances.values().map(|v| v.len()).sum::<usize>();
657        let code_size_factor = if generic_functions > 0 {
658            mono_instances as f64 / generic_functions as f64
659        } else {
660            1.0
661        };
662
663        Self {
664            generic_functions,
665            mono_instances,
666            unique_type_combos,
667            code_size_factor,
668        }
669    }
670}
