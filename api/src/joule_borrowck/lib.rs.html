1//! Borrow checker for Joule
2//!
3//! This module implements ownership and borrowing analysis using NLL (Non-Lexical Lifetimes).
4//!
5//! ## Features
6//!
7//! - **Use-after-move detection**: Catches use of values after they've been moved
8//! - **Use-of-uninitialized detection**: Ensures variables are initialized before use
9//! - **Mutability checking**: Prevents mutable borrows during existing borrows
10//! - **Lifetime tracking**: Region-based lifetime analysis for references
11//! - **Borrow conflict detection**: Detects conflicting borrows (multiple mutable, mixed mut/shared)
12//!
13//! ## Lifetime Model
14//!
15//! Joule uses a region-based lifetime system similar to Rust's NLL:
16//! - Each reference has an associated lifetime region
17//! - Regions are sets of program points where the borrow is active
18//! - Borrows must not outlive the borrowed data
19//! - Multiple shared borrows are allowed, but mutable borrows are exclusive
20
21pub mod lifetimes;

29/// State of a local variable with respect to ownership
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum LoanState {
32    /// Variable is declared but not initialized
33    Uninitialized,
34    /// Variable is initialized and can be used
35    Initialized,
36    /// Variable's value has been moved
37    Moved,
38    /// Variable is currently borrowed immutably
39    BorrowedShared,
40    /// Variable is currently borrowed mutably
41    BorrowedMut,
42    /// Variable is partially moved (some fields moved, others valid)
43    PartiallyMoved,
44    /// Variable is partially initialized (some fields initialized, others not)
45    PartiallyInitialized,
46}

48/// A path representing a place (local + projections)
49/// Used for tracking field-level borrows and moves
50#[derive(Debug, Clone, PartialEq, Eq, Hash)]
51pub struct PlacePath {
52    pub local: Local,
53    /// Field indices forming the path (e.g., [0, 1] for local.0.1)
54    pub fields: Vec<u32>,
55}
56
57impl PlacePath {
58    fn from_place(place: &Place) -> Self {
59        let fields = place
60            .projection
61            .iter()
62            .filter_map(|elem| match elem {
63                joule_mir::PlaceElem::Field(idx) => Some(idx.0),
64                joule_mir::PlaceElem::Deref
65                | joule_mir::PlaceElem::Index(_)
66                | joule_mir::PlaceElem::Downcast(_) => None,
67            })
68            .collect();
69
70        Self {
71            local: place.local,
72            fields,
73        }
74    }
75
76    /// Check if this path is a prefix of another (i.e., this place contains the other)
77    fn is_prefix_of(&self, other: &Self) -> bool {
78        if self.local != other.local {
79            return false;
80        }
81        if self.fields.len() > other.fields.len() {
82            return false;
83        }
84        self.fields
85            .iter()
86            .zip(other.fields.iter())
87            .all(|(a, b)| a == b)
88    }
89
90    /// Check if two paths conflict (one is prefix of the other, or they're the same)
91    fn conflicts_with(&self, other: &Self) -> bool {
92        if self.local != other.local {
93            return false;
94        }
95        self.is_prefix_of(other) || other.is_prefix_of(self)
96    }
97}

99/// Tracks the state of all locals in a function, including field-level tracking
100#[derive(Debug, Clone)]
101struct LocalStates {
102    /// State of root locals (without projection)
103    states: HashMap<Local, LoanState>,
104    /// State of specific field paths that have been moved
105    moved_paths: HashSet<PlacePath>,
106    /// State of specific field paths that are borrowed (shared)
107    borrowed_shared_paths: HashSet<PlacePath>,
108    /// State of specific field paths that are borrowed (mutable)
109    borrowed_mut_paths: HashSet<PlacePath>,
110}
111
112impl LocalStates {
113    fn new() -> Self {
114        Self {
115            states: HashMap::new(),
116            moved_paths: HashSet::new(),
117            borrowed_shared_paths: HashSet::new(),
118            borrowed_mut_paths: HashSet::new(),
119        }
120    }
121
122    fn get(&self, local: Local) -> LoanState {
123        self.states
124            .get(&local)
125            .copied()
126            .unwrap_or(LoanState::Uninitialized)
127    }
128
129    fn set(&mut self, local: Local, state: LoanState) {
130        self.states.insert(local, state);
131    }
132
133    /// Check if a path is moved (either directly or via parent being moved)
134    fn is_path_moved(&self, path: &PlacePath) -> bool {
135        // Check if this exact path is moved
136        if self.moved_paths.contains(path) {
137            return true;
138        }
139        // Check if any parent path is moved
140        for moved in &self.moved_paths {
141            if moved.is_prefix_of(path) {
142                return true;
143            }
144        }
145        false
146    }
147
148    /// Check if a path is borrowed (shared or mut)
149    fn is_path_borrowed(&self, path: &PlacePath) -> (bool, bool) {
150        let mut shared = false;
151        let mut mutable = false;
152
153        // Check for shared borrows that conflict
154        for borrowed in &self.borrowed_shared_paths {
155            if borrowed.conflicts_with(path) {
156                shared = true;
157            }
158        }
159
160        // Check for mutable borrows that conflict
161        for borrowed in &self.borrowed_mut_paths {
162            if borrowed.conflicts_with(path) {
163                mutable = true;
164            }
165        }
166
167        (shared, mutable)
168    }
169
170    /// Mark a path as moved
171    fn mark_moved(&mut self, path: PlacePath) {
172        self.moved_paths.insert(path.clone());
173        // If we moved the whole local, update root state
174        if path.fields.is_empty() {
175            self.set(path.local, LoanState::Moved);
176        } else {
177            // Partial move - mark root as partially moved if it was initialized
178            let current = self.get(path.local);
179            if current == LoanState::Initialized {
180                self.set(path.local, LoanState::PartiallyMoved);
181            }
182        }
183    }
184
185    /// Mark a path as borrowed
186    fn mark_borrowed(&mut self, path: PlacePath, mutable: bool) {
187        if mutable {
188            self.borrowed_mut_paths.insert(path.clone());
189        } else {
190            self.borrowed_shared_paths.insert(path.clone());
191        }
192        // Update root state
193        if path.fields.is_empty() {
194            self.set(
195                path.local,
196                if mutable {
197                    LoanState::BorrowedMut
198                } else {
199                    LoanState::BorrowedShared
200                },
201            );
202        }
203    }
204
205    /// Mark a path as initialized
206    fn mark_initialized(&mut self, path: &PlacePath) {
207        // Remove from moved set if present
208        self.moved_paths.remove(path);
209
210        if path.fields.is_empty() {
211            self.set(path.local, LoanState::Initialized);
212        } else {
213            // Partial initialization - check if all fields are now initialized
214            let current = self.get(path.local);
215            if current == LoanState::Uninitialized || current == LoanState::PartiallyMoved {
216                self.set(path.local, LoanState::PartiallyInitialized);
217            }
218        }
219    }
220
221    fn merge(&mut self, other: &Self) -> bool {
222        let mut changed = false;
223        for (&local, &other_state) in &other.states {
224            let current_state = self.get(local);
225            let merged_state = merge_loan_states(current_state, other_state);
226            if merged_state != current_state {
227                self.set(local, merged_state);
228                changed = true;
229            }
230        }
231        // Merge moved paths
232        for path in &other.moved_paths {
233            if self.moved_paths.insert(path.clone()) {
234                changed = true;
235            }
236        }
237        // Merge borrowed paths (conservatively keep all borrows)
238        for path in &other.borrowed_shared_paths {
239            if self.borrowed_shared_paths.insert(path.clone()) {
240                changed = true;
241            }
242        }
243        for path in &other.borrowed_mut_paths {
244            if self.borrowed_mut_paths.insert(path.clone()) {
245                changed = true;
246            }
247        }
248        changed
249    }
250}

252/// Merge two loan states from different control flow paths
253fn merge_loan_states(a: LoanState, b: LoanState) -> LoanState {
254    use LoanState::{
255        BorrowedMut, BorrowedShared, Initialized, Moved, PartiallyInitialized, PartiallyMoved,
256        Uninitialized,
257    };
258    match (a, b) {
259        // If both paths have the same state, keep it
260        (x, y) if x == y => x,
261
262        // If one path moved and the other didn't initialize, it's moved
263        (Moved, Uninitialized) | (Uninitialized, Moved) => Moved,
264
265        // If one path initialized and the other moved, conservative: moved
266        (Initialized, Moved) | (Moved, Initialized) => Moved,
267
268        // Partial states merge to partial
269        (PartiallyMoved, Initialized) | (Initialized, PartiallyMoved) => PartiallyMoved,
270        (PartiallyMoved, Moved) | (Moved, PartiallyMoved) => Moved,
271        (PartiallyInitialized, Uninitialized) | (Uninitialized, PartiallyInitialized) => {
272            PartiallyInitialized
273        }
274        (PartiallyInitialized, Initialized) | (Initialized, PartiallyInitialized) => {
275            PartiallyInitialized
276        }
277
278        // If one path didn't initialize, conservative: uninitialized
279        (Uninitialized, _) | (_, Uninitialized) => Uninitialized,
280
281        // If borrowed in different ways, conservative: moved
282        (BorrowedShared, BorrowedMut) | (BorrowedMut, BorrowedShared) => Moved,
283
284        // Default: moved (most conservative)
285        _ => Moved,
286    }
287}

313    /// Run the borrow checker on a MIR context (all functions)
314    pub fn check_context(&mut self, ctx: &joule_mir::MirContext) -> Result<(), Vec<Diagnostic>> {
315        for (_, function) in &ctx.functions {
316            self.check_function(function);
317        }
318
319        if self.diagnostics.is_empty() {
320            Ok(())
321        } else {
322            Err(std::mem::take(&mut self.diagnostics))
323        }
324    }

326    /// Run the borrow checker on a function
327    pub fn check(&mut self, function: &FunctionMIR) -> Result<(), Vec<Diagnostic>> {
328        self.check_function(function);
329
330        if self.diagnostics.is_empty() {
331            Ok(())
332        } else {
333            Err(std::mem::take(&mut self.diagnostics))
334        }
335    }

337    /// Check a single function
338    fn check_function(&mut self, function: &FunctionMIR) {
339        // Initialize lifetime context for this function
340        self.lifetime_ctx = Some(LifetimeContext::new());
341
342        // Initialize entry block state
343        let mut block_entry_states: HashMap<BasicBlockId, LocalStates> = HashMap::new();
344
345        let mut entry_state = LocalStates::new();
346        // Function parameters are initialized
347        for &param in &function.params {
348            entry_state.set(param, LoanState::Initialized);
349        }
350
351        block_entry_states.insert(BasicBlockId::START, entry_state);
352
353        // Worklist algorithm for dataflow analysis
354        let mut worklist = VecDeque::new();
355        worklist.push_back(BasicBlockId::START);
356        let mut visited = HashSet::new();
357
358        while let Some(block_id) = worklist.pop_front() {
359            if visited.contains(&block_id) {
360                continue;
361            }
362            visited.insert(block_id);
363
364            let entry_state = block_entry_states
365                .get(&block_id)
366                .cloned()
367                .unwrap_or_else(LocalStates::new);
368
369            let block = &function.basic_blocks[block_id.0 as usize];
370            let mut state = entry_state.clone();
371
372            // Check all statements in the block
373            for (stmt_idx, stmt) in block.statements.iter().enumerate() {
374                // Update current program point for lifetime tracking
375                self.current_point = Some(ProgramPoint::new(block_id, stmt_idx as u32));
376                self.check_statement(stmt, &mut state);
377            }
378
379            // Check terminator
380            self.check_terminator(&block.terminator, &mut state, &function.return_ty);
381
382            // Propagate state to successor blocks
383            match &block.terminator {
384                Terminator::Return { .. } => {
385                    // No successors
386                }
387                Terminator::Goto { target, .. } => {
388                    let changed =
389                        self.merge_state_into(target, state.clone(), &mut block_entry_states);
390                    if changed {
391                        worklist.push_back(*target);
392                    }
393                }
394                Terminator::Call { target, .. } => {
395                    let changed =
396                        self.merge_state_into(target, state.clone(), &mut block_entry_states);
397                    if changed {
398                        worklist.push_back(*target);
399                    }
400                }
401                Terminator::SwitchInt { targets, .. } => {
402                    // Iterate over all branches
403                    for &(_value, target) in &targets.branches {
404                        let changed =
405                            self.merge_state_into(&target, state.clone(), &mut block_entry_states);
406                        if changed {
407                            worklist.push_back(target);
408                        }
409                    }
410                    // Also handle the otherwise branch
411                    let changed = self.merge_state_into(
412                        &targets.otherwise,
413                        state.clone(),
414                        &mut block_entry_states,
415                    );
416                    if changed {
417                        worklist.push_back(targets.otherwise);
418                    }
419                }
420                Terminator::Unreachable { .. }
421                | Terminator::Abort { .. }
422                | Terminator::Cancel { .. } => {
423                    // No successors
424                }
425
426                // Concurrency terminators
427                Terminator::Spawn { target, .. }
428                | Terminator::TaskAwait { target, .. }
429                | Terminator::TaskGroupExit { target, .. } => {
430                    let changed =
431                        self.merge_state_into(target, state.clone(), &mut block_entry_states);
432                    if changed {
433                        worklist.push_back(*target);
434                    }
435                }
436                Terminator::TaskGroupEnter {
437                    body, join_block, ..
438                } => {
439                    let changed1 =
440                        self.merge_state_into(body, state.clone(), &mut block_entry_states);
441                    let changed2 =
442                        self.merge_state_into(join_block, state.clone(), &mut block_entry_states);
443                    if changed1 {
444                        worklist.push_back(*body);
445                    }
446                    if changed2 {
447                        worklist.push_back(*join_block);
448                    }
449                }
450                Terminator::ChannelRecv {
451                    target,
452                    closed_target,
453                    ..
454                }
455                | Terminator::ChannelSend {
456                    target,
457                    closed_target,
458                    ..
459                } => {
460                    let changed1 =
461                        self.merge_state_into(target, state.clone(), &mut block_entry_states);
462                    let changed2 = self.merge_state_into(
463                        closed_target,
464                        state.clone(),
465                        &mut block_entry_states,
466                    );
467                    if changed1 {
468                        worklist.push_back(*target);
469                    }
470                    if changed2 {
471                        worklist.push_back(*closed_target);
472                    }
473                }
474                Terminator::Select { arms, default, .. } => {
475                    for arm in arms {
476                        let changed = self.merge_state_into(
477                            &arm.target,
478                            state.clone(),
479                            &mut block_entry_states,
480                        );
481                        if changed {
482                            worklist.push_back(arm.target);
483                        }
484                    }
485                    if let Some(default) = default {
486                        let changed =
487                            self.merge_state_into(default, state.clone(), &mut block_entry_states);
488                        if changed {
489                            worklist.push_back(*default);
490                        }
491                    }
492                }
493            }
494        }
495
496        // Solve lifetime constraints and collect any errors
497        if let Some(ref mut lifetime_ctx) = self.lifetime_ctx
498            && let Err(lifetime_errors) = lifetime_ctx.solve_constraints()
499        {
500            for err in lifetime_errors {
501                self.diagnostics.push(err.to_diagnostic());
502            }
503        }
504
505        // Clear lifetime context for next function
506        self.lifetime_ctx = None;
507        self.current_point = None;
508    }

510    fn merge_state_into(
511        &self,
512        target: &BasicBlockId,
513        state: LocalStates,
514        block_states: &mut HashMap<BasicBlockId, LocalStates>,
515    ) -> bool {
516        if let Some(existing) = block_states.get_mut(target) {
517            existing.merge(&state)
518        } else {
519            block_states.insert(*target, state);
520            true
521        }
522    }

524    fn check_statement(&mut self, stmt: &Statement, state: &mut LocalStates) {
525        match stmt {
526            Statement::Assign {
527                place,
528                rvalue,
529                span,
530            } => {
531                // Check that RHS is valid and update state for moves
532                self.check_rvalue(rvalue, state, *span);
533
534                // LHS is now initialized - use field-level tracking
535                let path = self.place_to_path(place);
536                state.mark_initialized(&path);
537            }
538            Statement::StorageLive { local, .. } => {
539                // Mark as uninitialized (storage allocated but no value)
540                state.set(*local, LoanState::Uninitialized);
541            }
542            Statement::StorageDead { local, .. } => {
543                // Mark as moved (can't use after storage dead)
544                state.set(*local, LoanState::Moved);
545                // Also clear any field-level tracking for this local
546                state.moved_paths.retain(|p| p.local != *local);
547                state.borrowed_shared_paths.retain(|p| p.local != *local);
548                state.borrowed_mut_paths.retain(|p| p.local != *local);
549            }
550            Statement::Nop => {}
551        }
552    }

554    fn check_rvalue(&mut self, rvalue: &Rvalue, state: &mut LocalStates, span: joule_common::Span) {
555        match rvalue {
556            Rvalue::Use(operand) => {
557                self.check_operand(operand, state, false, span);
558            }
559            Rvalue::Ref { place, mutable } => {
560                // First check basic place validity
561                self.check_place(place, state, *mutable, span);
562
563                // Then perform lifetime-based borrow checking
564                if let (Some(lifetime_ctx), Some(point)) =
565                    (&mut self.lifetime_ctx, self.current_point)
566                {
567                    let local = place.local;
568
569                    // Check if borrow is allowed at this point
570                    let borrow_result = if *mutable {
571                        lifetime_ctx.can_borrow_mut(local, point)
572                    } else {
573                        lifetime_ctx.can_borrow_shared(local, point)
574                    };
575
576                    // Record any borrow conflicts
577                    if let Err(err) = borrow_result {
578                        self.diagnostics.push(err.to_diagnostic());
579                    }
580
581                    // Create the loan (borrow) for lifetime tracking
582                    lifetime_ctx.create_loan(place.clone(), *mutable, point, span);
583
584                    // Track the borrow path precisely for conflict detection.
585                    // mark_borrowed records both the path-level borrow state and
586                    // updates the root local's LoanState.
587                    let path = self.place_to_path(place);
588                    state.mark_borrowed(path, *mutable);
589                }
590            }
591            Rvalue::BinaryOp { left, right, .. } => {
592                self.check_operand(left, state, false, span);
593                self.check_operand(right, state, false, span);
594            }
595            Rvalue::UnaryOp { operand, .. } => {
596                self.check_operand(operand, state, false, span);
597            }
598            Rvalue::Discriminant { place } => {
599                self.check_place(place, state, false, span);
600            }
601            Rvalue::Aggregate { operands, .. } => {
602                for operand in operands {
603                    self.check_operand(operand, state, false, span);
604                }
605            }
606            Rvalue::Cast { operand, .. } => {
607                self.check_operand(operand, state, false, span);
608            }
609            Rvalue::Len { place } => {
610                self.check_place(place, state, false, span);
611            }
612            // SIMD operations - check their operands and places
613            Rvalue::SimdBinaryOp { left, right, .. } => {
614                self.check_operand(left, state, false, span);
615                self.check_operand(right, state, false, span);
616            }
617            Rvalue::SimdLoad { source, .. } => {
618                self.check_place(source, state, false, span);
619            }
620            Rvalue::SimdStore { value, dest, .. } => {
621                self.check_operand(value, state, false, span);
622                self.check_place(dest, state, true, span); // Store is a mutable operation
623            }
624            Rvalue::SimdSplat { value, .. } => {
625                self.check_operand(value, state, false, span);
626            }
627
628            // Concurrency rvalues
629            Rvalue::ChannelCreate { .. } | Rvalue::IsCancelled | Rvalue::CurrentTask => {
630                // No operands to check
631            }
632            Rvalue::ChannelTryRecv { channel }
633            | Rvalue::ChannelSender { channel }
634            | Rvalue::ChannelReceiver { channel }
635            | Rvalue::ChannelClose { channel } => {
636                self.check_operand(channel, state, false, span);
637            }
638            Rvalue::ChannelTrySend { channel, value } => {
639                self.check_operand(channel, state, false, span);
640                self.check_operand(value, state, false, span);
641            }
642            Rvalue::Try(operand) => {
643                self.check_operand(operand, state, false, span);
644            }
645            Rvalue::EnumVariant { fields, .. } => {
646                for op in fields {
647                    self.check_operand(op, state, false, span);
648                }
649            }
650        }
651    }

653    fn check_place(
654        &mut self,
655        place: &Place,
656        state: &LocalStates,
657        is_mut_borrow: bool,
658        span: joule_common::Span,
659    ) {
660        let path = self.place_to_path(place);
661        let local = self.place_to_local(place).expect("place must have a local");
662        let loan_state = state.get(local);
663
664        // Places with dereferences may alias other borrows; record for diagnostics
665        let _has_deref = self.place_has_deref(place);
666
667        // Check for use of uninitialized
668        match loan_state {
669            LoanState::Uninitialized => {
670                self.diagnostics.push(
671                    Diagnostic::error("use of possibly-uninitialized variable")
672                        .with_label(Label::primary(span, "used here"))
673                        .with_code("E0381"),
674                );
675                return;
676            }
677            LoanState::PartiallyInitialized if path.fields.is_empty() => {
678                self.diagnostics.push(
679                    Diagnostic::error("use of partially-initialized value")
680                        .with_label(Label::primary(span, "not all fields are initialized"))
681                        .with_code("E0381"),
682                );
683                return;
684            }
685            _ => {}
686        }
687
688        // Check for use of moved value using field-level tracking
689        if state.is_path_moved(&path) {
690            self.diagnostics.push(
691                Diagnostic::error("use of moved value")
692                    .with_label(Label::primary(span, "value used here after move"))
693                    .with_code("E0382"),
694            );
695            return;
696        }
697
698        // Check for use of whole value when it's partially moved
699        if path.fields.is_empty() && loan_state == LoanState::PartiallyMoved {
700            self.diagnostics.push(
701                Diagnostic::error("use of partially-moved value")
702                    .with_label(Label::primary(span, "some fields have been moved"))
703                    .with_code("E0382"),
704            );
705            return;
706        }
707
708        // Check borrow conflicts
709        let (has_shared, has_mut) = state.is_path_borrowed(&path);
710        if is_mut_borrow && (has_shared || has_mut) {
711            self.diagnostics.push(
712                Diagnostic::error(
713                    "cannot borrow as mutable because it is also borrowed as immutable",
714                )
715                .with_label(Label::primary(span, "mutable borrow occurs here"))
716                .with_code("E0502"),
717            );
718        }
719    }

721    fn check_operand(
722        &mut self,
723        operand: &Operand,
724        state: &mut LocalStates,
725        is_mut_borrow: bool,
726        span: joule_common::Span,
727    ) {
728        match operand {
729            Operand::Move(place) => {
730                let path = self.place_to_path(place);
731                let local = place.local;
732                let loan_state = state.get(local);
733
734                // Check for use of uninitialized
735                match loan_state {
736                    LoanState::Uninitialized => {
737                        self.diagnostics.push(
738                            Diagnostic::error("use of possibly-uninitialized variable")
739                                .with_label(Label::primary(span, "used here"))
740                                .with_code("E0381"),
741                        );
742                        return;
743                    }
744                    LoanState::PartiallyInitialized if path.fields.is_empty() => {
745                        self.diagnostics.push(
746                            Diagnostic::error("cannot move partially-initialized value")
747                                .with_label(Label::primary(span, "not all fields are initialized"))
748                                .with_code("E0381"),
749                        );
750                        return;
751                    }
752                    _ => {}
753                }
754
755                // Check for use of moved value using field-level tracking
756                if state.is_path_moved(&path) {
757                    self.diagnostics.push(
758                        Diagnostic::error("use of moved value")
759                            .with_label(Label::primary(span, "value used here after move"))
760                            .with_code("E0382"),
761                    );
762                    return;
763                }
764
765                // Check for moving out of partially moved value
766                if path.fields.is_empty() && loan_state == LoanState::PartiallyMoved {
767                    self.diagnostics.push(
768                        Diagnostic::error("cannot move out of partially-moved value")
769                            .with_label(Label::primary(span, "some fields have been moved"))
770                            .with_code("E0382"),
771                    );
772                    return;
773                }
774
775                // Check for borrow conflicts
776                let (has_shared, has_mut) = state.is_path_borrowed(&path);
777                if has_shared || has_mut {
778                    // Check via lifetime context for precise NLL analysis
779                    if let (Some(lifetime_ctx), Some(point)) =
780                        (&self.lifetime_ctx, self.current_point)
781                    {
782                        if let Err(err) = lifetime_ctx.can_move(local, point) {
783                            self.diagnostics.push(err.to_diagnostic());
784                            return;
785                        }
786                        // Lifetime context says move is OK (borrow ended)
787                    } else {
788                        // No lifetime context, use conservative check
789                        self.diagnostics.push(
790                            Diagnostic::error("cannot move out of borrowed content")
791                                .with_label(Label::primary(span, "cannot move while borrowed"))
792                                .with_code("E0505"),
793                        );
794                        return;
795                    }
796                }
797
798                // Also verify via lifetime context if available
799                if let (Some(lifetime_ctx), Some(point)) = (&self.lifetime_ctx, self.current_point)
800                    && let Err(err) = lifetime_ctx.can_move(local, point)
801                {
802                    self.diagnostics.push(err.to_diagnostic());
803                    return;
804                }
805
806                // OK, mark as moved using field-level tracking
807                state.mark_moved(path);
808            }
809            Operand::Copy(place) => {
810                let path = self.place_to_path(place);
811                let local = place.local;
812                let loan_state = state.get(local);
813
814                // Check for use of uninitialized
815                match loan_state {
816                    LoanState::Uninitialized => {
817                        self.diagnostics.push(
818                            Diagnostic::error("use of possibly-uninitialized variable")
819                                .with_label(Label::primary(span, "used here"))
820                                .with_code("E0381"),
821                        );
822                        return;
823                    }
824                    LoanState::PartiallyInitialized if path.fields.is_empty() => {
825                        self.diagnostics.push(
826                            Diagnostic::error("cannot copy partially-initialized value")
827                                .with_label(Label::primary(span, "not all fields are initialized"))
828                                .with_code("E0381"),
829                        );
830                        return;
831                    }
832                    _ => {}
833                }
834
835                // Check for use of moved value using field-level tracking
836                if state.is_path_moved(&path) {
837                    self.diagnostics.push(
838                        Diagnostic::error("use of moved value")
839                            .with_label(Label::primary(span, "value used here after move"))
840                            .with_code("E0382"),
841                    );
842                    return;
843                }
844
845                // Check for copying from partially moved value
846                if path.fields.is_empty() && loan_state == LoanState::PartiallyMoved {
847                    self.diagnostics.push(
848                        Diagnostic::error("cannot copy partially-moved value")
849                            .with_label(Label::primary(span, "some fields have been moved"))
850                            .with_code("E0382"),
851                    );
852                    return;
853                }
854
855                // Check borrow conflicts for mutable borrows
856                let (_, has_mut) = state.is_path_borrowed(&path);
857                if has_mut && is_mut_borrow {
858                    self.diagnostics.push(
859                        Diagnostic::error("cannot borrow as mutable more than once")
860                            .with_label(Label::primary(span, "second mutable borrow"))
861                            .with_code("E0499"),
862                    );
863                }
864                // Copy is OK if initialized or shared-borrowed
865            }
866            Operand::Constant(_) => {
867                // Constants are always OK
868            }
869        }
870    }

872    fn check_terminator(
873        &mut self,
874        terminator: &Terminator,
875        state: &mut LocalStates,
876        return_ty: &joule_mir::Ty,
877    ) {
878        match terminator {
879            Terminator::Return { span } => {
880                // Unit-returning functions don't assign to the return place — skip the check
881                if matches!(return_ty, joule_mir::Ty::Unit) {
882                    // Nothing to verify — unit functions have no return value
883                } else {
884                    let ret_state = state.get(Local::RETURN);
885                    match ret_state {
886                        LoanState::Initialized
887                        | LoanState::BorrowedShared
888                        | LoanState::BorrowedMut => {
889                            // OK — return value has been assigned
890                        }
891                        LoanState::Uninitialized => {
892                            self.diagnostics.push(
893                                Diagnostic::error("use of possibly-uninitialized return value")
894                                    .with_label(Label::primary(
895                                        *span,
896                                        "returning here without initializing return value",
897                                    ))
898                                    .with_code("E0381"),
899                            );
900                        }
901                        LoanState::Moved => {
902                            self.diagnostics.push(
903                                Diagnostic::error("return value has been moved")
904                                    .with_label(Label::primary(
905                                        *span,
906                                        "return value was moved before return",
907                                    ))
908                                    .with_code("E0382"),
909                            );
910                        }
911                        LoanState::PartiallyMoved => {
912                            self.diagnostics.push(
913                                Diagnostic::error("return value is partially moved")
914                                    .with_label(Label::primary(
915                                        *span,
916                                        "some fields of return value have been moved",
917                                    ))
918                                    .with_code("E0382"),
919                            );
920                        }
921                        LoanState::PartiallyInitialized => {
922                            self.diagnostics.push(
923                                Diagnostic::error("return value is only partially initialized")
924                                    .with_label(Label::primary(
925                                        *span,
926                                        "not all fields of return value are initialized",
927                                    ))
928                                    .with_code("E0381"),
929                            );
930                        }
931                    }
932                }
933            }
934            Terminator::Goto { .. } => {}
935            Terminator::Call { args, span, .. } => {
936                for arg in args {
937                    self.check_operand(arg, state, false, *span);
938                }
939            }
940            Terminator::SwitchInt {
941                discriminant, span, ..
942            } => {
943                self.check_operand(discriminant, state, false, *span);
944            }
945            Terminator::Unreachable { .. }
946            | Terminator::Abort { .. }
947            | Terminator::Cancel { .. } => {}
948
949            // Concurrency terminators
950            Terminator::Spawn {
951                func, args, span, ..
952            } => {
953                self.check_operand(func, state, false, *span);
954                for arg in args {
955                    self.check_operand(arg, state, false, *span);
956                }
957            }
958            Terminator::TaskAwait { task, span, .. } => {
959                self.check_operand(task, state, false, *span);
960            }
961            Terminator::TaskGroupEnter { .. } | Terminator::TaskGroupExit { group: _, .. } => {
962                // Group handle is checked through regular operand flow
963            }
964            Terminator::ChannelRecv { channel, span, .. } => {
965                self.check_operand(channel, state, false, *span);
966            }
967            Terminator::ChannelSend {
968                channel,
969                value,
970                span,
971                ..
972            } => {
973                self.check_operand(channel, state, false, *span);
974                self.check_operand(value, state, false, *span);
975            }
976            Terminator::Select { arms, span, .. } => {
977                for arm in arms {
978                    match &arm.operation {
979                        joule_mir::ChannelOp::Recv { channel } => {
980                            self.check_operand(channel, state, false, *span);
981                        }
982                        joule_mir::ChannelOp::Send { channel, value } => {
983                            self.check_operand(channel, state, false, *span);
984                            self.check_operand(value, state, false, *span);
985                        }
986                        joule_mir::ChannelOp::Timeout { .. } => {}
987                    }
988                }
989            }
990        }
991    }
