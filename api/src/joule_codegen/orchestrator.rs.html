10/// Compilation options
11#[derive(Debug, Clone)]
12pub struct CompileOptions {
13    /// Requested backend (None = auto-select)
14    pub backend: Option<Backend>,
15    /// Optimization level (0-3)
16    pub opt_level: u8,
17    /// Generate debug info
18    pub debug_info: bool,
19    /// Enable LTO
20    pub lto: bool,
21    /// Target specification
22    pub target: Target,
23    /// Emit intermediate representations
24    pub emit_ir: bool,
25    /// Prefer compile speed over optimization
26    pub prefer_speed: bool,
27}
28
29impl CompileOptions {
30    /// Create options for debug builds (fast compilation)
31    pub fn debug() -> Self {
32        Self {
33            backend: Some(Backend::Cranelift),
34            opt_level: 0,
35            debug_info: true,
36            lto: false,
37            target: Target::native(),
38            emit_ir: false,
39            prefer_speed: true,
40        }
41    }
42
43    /// Create options for release builds (optimized)
44    pub fn release() -> Self {
45        Self {
46            backend: Some(Backend::Llvm),
47            opt_level: 3,
48            debug_info: false,
49            lto: true,
50            target: Target::native(),
51            emit_ir: false,
52            prefer_speed: false,
53        }
54    }
55
56    /// Create options for GPU/heterogeneous builds
57    pub fn heterogeneous() -> Self {
58        Self {
59            backend: Some(Backend::Mlir),
60            opt_level: 2,
61            debug_info: false,
62            lto: false,
63            target: Target::native(),
64            emit_ir: false,
65            prefer_speed: false,
66        }
67    }
68
69    /// Create options with auto backend selection
70    pub fn auto() -> Self {
71        Self {
72            backend: None,
73            opt_level: 2,
74            debug_info: true,
75            lto: false,
76            target: Target::native(),
77            emit_ir: false,
78            prefer_speed: false,
79        }
80    }
81}

113/// The backend orchestrator
114///
115/// Manages multiple codegen backends and selects the best one for each compilation.
116pub struct Orchestrator {
117    /// Available backends
118    available: Vec<Backend>,
119}
120
121impl Orchestrator {
122    /// Create a new orchestrator
123    pub fn new() -> Self {
124        let mut available = Vec::new();
125
126        // Check which backends are available
127        if cfg!(feature = "cranelift") {
128            available.push(Backend::Cranelift);
129        }
130        if cfg!(feature = "llvm") {
131            available.push(Backend::Llvm);
132        }
133        if cfg!(feature = "mlir") {
134            available.push(Backend::Mlir);
135        }
136
137        // Always have at least Cranelift available as fallback
138        if available.is_empty() {
139            available.push(Backend::Cranelift);
140        }
141
142        Self { available }
143    }
144
145    /// Get list of available backends
146    pub fn available_backends(&self) -> &[Backend] {
147        &self.available
148    }
149
150    /// Select the best backend for the given options and code
151    pub fn select_backend(
152        &self,
153        mir: &MirContext,
154        options: &CompileOptions,
155    ) -> CodegenResult<Backend> {
156        // If user specified a backend, use it
157        if let Some(backend) = options.backend {
158            if self.available.contains(&backend) {
159                return Ok(backend);
160            } else {
161                return Err(CodegenError::BackendUnavailable {
162                    backend: backend.name().to_string(),
163                    reason: "not compiled into this build".to_string(),
164                });
165            }
166        }
167
168        // Auto-select based on code characteristics and options
169        let needs_gpu = self.code_needs_gpu(mir);
170        let prefer_speed = options.prefer_speed;
171        let high_opt = options.opt_level >= 2;
172
173        debug!(
174            "Auto-selecting backend: needs_gpu={}, prefer_speed={}, high_opt={}",
175            needs_gpu, prefer_speed, high_opt
176        );
177
178        // Decision tree:
179        // 1. If code needs GPU and MLIR is available, use MLIR
180        // 2. If prefer speed and Cranelift available, use Cranelift
181        // 3. If high optimization and LLVM available, use LLVM
182        // 4. Fall back to first available backend
183
184        if needs_gpu && self.available.contains(&Backend::Mlir) {
185            info!("Selected MLIR backend for GPU code");
186            return Ok(Backend::Mlir);
187        }
188
189        if prefer_speed && self.available.contains(&Backend::Cranelift) {
190            info!("Selected Cranelift backend for fast compilation");
191            return Ok(Backend::Cranelift);
192        }
193
194        if high_opt && self.available.contains(&Backend::Llvm) {
195            info!("Selected LLVM backend for high optimization");
196            return Ok(Backend::Llvm);
197        }
198
199        // Default to first available
200        self.available
201            .first()
202            .copied()
203            .ok_or(CodegenError::NoSuitableBackend)
204    }
205
206    /// Check if the code needs GPU support by inspecting function attributes
207    /// and MIR operations.
208    ///
209    /// Returns true if any function:
210    /// - Has the `#[gpu]` or `#[gpu_kernel]` attribute
211    /// - Has a `target` attribute set to `ProcessorTarget::Gpu`
212    /// - Contains a raw attribute with "gpu" or "parallel" markers
213    /// - Contains SIMD operations that benefit from GPU execution
214    fn code_needs_gpu(&self, mir: &MirContext) -> bool {
215        use joule_mir::{Rvalue, Statement};
216
217        for (_id, func) in &mir.functions {
218            // Check the is_gpu_kernel flag (set by #[gpu] or #[gpu_kernel])
219            if func.is_gpu_kernel() {
220                return true;
221            }
222
223            // Check raw attributes for GPU-related markers. These cover
224            // user-defined attributes like #[compute_shader], #[gpu_dispatch],
225            // or #[parallel(gpu)].
226            for (attr_name, attr_args) in &func.attributes.raw {
227                let name_lower = attr_name.to_lowercase();
228                if name_lower.contains("gpu")
229                    || name_lower.contains("compute_shader")
230                    || name_lower.contains("kernel")
231                {
232                    return true;
233                }
234                // Check for #[parallel(gpu)] or #[target(gpu)]
235                for (key, value) in attr_args {
236                    let key_lower = key.to_lowercase();
237                    let value_lower = value.to_lowercase();
238                    if key_lower == "gpu" || value_lower == "gpu" {
239                        return true;
240                    }
241                }
242            }
243
244            // Check for heavy SIMD usage that would benefit from GPU offload.
245            // Functions with SIMD operations across multiple basic blocks
246            // indicate data-parallel workloads suitable for GPU execution.
247            let mut simd_op_count = 0;
248            for block in &func.basic_blocks {
249                for stmt in &block.statements {
250                    if let Statement::Assign { rvalue, .. } = stmt {
251                        match rvalue {
252                            Rvalue::SimdBinaryOp { .. }
253                            | Rvalue::SimdLoad { .. }
254                            | Rvalue::SimdStore { .. }
255                            | Rvalue::SimdSplat { .. } => {
256                                simd_op_count += 1;
257                            }
258                            _ => {}
259                        }
260                    }
261                }
262            }
263            // Threshold: functions with significant SIMD usage (10+ ops)
264            // are candidates for GPU offload when the MLIR backend is available
265            if simd_op_count >= 10 {
266                debug!(
267                    "Function '{}' has {} SIMD operations, GPU candidate",
268                    func.name.as_str(),
269                    simd_op_count
270                );
271                return true;
272            }
273        }
274        false
275    }

277    /// Compile MIR to object code
278    pub fn compile(
279        &self,
280        mir: &MirContext,
281        options: CompileOptions,
282    ) -> CodegenResult<CompileResult> {
283        let backend = self.select_backend(mir, &options)?;
284        self.compile_with_backend(mir, backend, options)
285    }
286
287    /// Compile with a specific backend
288    pub fn compile_with_backend(
289        &self,
290        mir: &MirContext,
291        backend: Backend,
292        options: CompileOptions,
293    ) -> CodegenResult<CompileResult> {
294        let start = std::time::Instant::now();
295
296        info!(
297            "Compiling with {} backend (opt_level={})",
298            backend, options.opt_level
299        );
300
301        let (object_code, ir_text) = match backend {
302            Backend::Cranelift => {
303                let codegen =
304                    joule_codegen_cranelift::CraneliftCodegen::new(None).map_err(|e| {
305                        CodegenError::CompilationFailed {
306                            backend: "cranelift".to_string(),
307                            message: e.to_string(),
308                        }
309                    })?;
310                let object_bytes =
311                    codegen
312                        .compile(mir)
313                        .map_err(|e| CodegenError::CompilationFailed {
314                            backend: "cranelift".to_string(),
315                            message: e.to_string(),
316                        })?;
317                (Some(object_bytes), None)
318            }
319            Backend::Llvm => {
320                #[cfg(feature = "llvm")]
321                {
322                    use joule_codegen_llvm::LLVMCodegen;
323
324                    let context = inkwell::context::Context::create();
325                    let mut codegen = LLVMCodegen::new(&context, "joule_module");
326                    codegen
327                        .compile_context(mir)
328                        .map_err(|e| CodegenError::CompilationFailed {
329                            backend: "llvm".to_string(),
330                            message: e.to_string(),
331                        })?;
332
333                    // Write to temp file and read back
334                    let temp_path =
335                        std::env::temp_dir().join(format!("joule_llvm_{}.o", std::process::id()));
336                    codegen.write_object_file(&temp_path).map_err(|e| {
337                        CodegenError::CompilationFailed {
338                            backend: "llvm".to_string(),
339                            message: e.to_string(),
340                        }
341                    })?;
342                    let object_bytes = std::fs::read(&temp_path)?;
343                    let _ = std::fs::remove_file(&temp_path);
344                    (Some(object_bytes), None)
345                }
346                #[cfg(not(feature = "llvm"))]
347                {
348                    return Err(CodegenError::BackendUnavailable {
349                        backend: "llvm".to_string(),
350                        reason: "not compiled into this build".to_string(),
351                    });
352                }
353            }
354            Backend::Mlir => {
355                #[cfg(feature = "mlir")]
356                {
357                    use joule_codegen_mlir::{JouleContext, MlirCodegen, MlirTarget};
358
359                    let context = JouleContext::new().map_err(CodegenError::MlirError)?;
360                    let codegen =
361                        MlirCodegen::new(MlirTarget::Cpu).with_opt_level(options.opt_level);
362
363                    let ir_text = if options.emit_ir {
364                        let text = codegen
365                            .compile_to_mlir_text(&context, mir)
366                            .map_err(CodegenError::MlirError)?;
367                        Some(text)
368                    } else {
369                        None
370                    };
371
372                    let object_bytes = codegen
373                        .compile(&context, mir)
374                        .map_err(CodegenError::MlirError)?;
375
376                    (Some(object_bytes), ir_text)
377                }
378                #[cfg(not(feature = "mlir"))]
379                {
380                    return Err(CodegenError::BackendUnavailable {
381                        backend: "mlir".to_string(),
382                        reason: "not compiled into this build".to_string(),
383                    });
384                }
385            }
386        };
387
388        let elapsed = start.elapsed();
389        let code_size_estimate = object_code.as_ref().map(|c| c.len()).unwrap_or(0);
390
391        Ok(CompileResult {
392            backend,
393            object_code,
394            ir_text,
395            stats: CompileStats {
396                compile_time_us: elapsed.as_micros() as u64,
397                functions_compiled: mir.functions.len(),
398                code_size_estimate,
399            },
400        })
401    }

414    #[test]
415    fn test_orchestrator_new() {
416        let orch = Orchestrator::new();
417        assert!(!orch.available_backends().is_empty());
418    }

420    #[test]
421    fn test_compile_options_presets() {
422        let debug = CompileOptions::debug();
423        assert_eq!(debug.opt_level, 0);
424        assert!(debug.debug_info);
425
426        let release = CompileOptions::release();
427        assert_eq!(release.opt_level, 3);
428        assert!(release.lto);
429
430        let hetero = CompileOptions::heterogeneous();
431        assert_eq!(hetero.backend, Some(Backend::Mlir));
432    }
433
434    #[test]
435    fn test_backend_selection_explicit() {
436        let orch = Orchestrator::new();
437        let mir = MirContext::new();
438        let options = CompileOptions {
439            backend: Some(Backend::Cranelift),
440            ..CompileOptions::default()
441        };
442
443        let backend = orch.select_backend(&mir, &options);
444        // Should return Cranelift if available, or error if not
445        if orch.available_backends().contains(&Backend::Cranelift) {
446            assert_eq!(backend.unwrap(), Backend::Cranelift);
447        }
448    }

450    #[test]

452        let orch = Orchestrator::new();
453        let mir = MirContext::new();
454        let options = CompileOptions::auto();
455
456        let backend = orch.select_backend(&mir, &options);
457        assert!(backend.is_ok());
458    }
