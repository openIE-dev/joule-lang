1//! Backend Orchestrator for Joule Compiler
2//!
3//! This crate provides unified access to all Joule codegen backends:
4//! - Cranelift: Fast compilation for debug builds
5//! - LLVM: Optimized compilation for release builds
6//! - MLIR: Heterogeneous computing (GPU, accelerators)
7//!
8//! # Backend Selection
9//!
10//! The orchestrator can automatically select the best backend based on:
11//! - Build profile (debug vs release)
12//! - Target architecture
13//! - Code characteristics (contains GPU kernels, etc.)
14//! - User preferences (--backend flag)
15//!
16//! # Hybrid Compilation
17//!
18//! For programs with mixed CPU and GPU code, the hybrid compiler can:
19//! - Analyze MIR to detect GPU-annotated functions
20//! - Split code into CPU and GPU modules
21//! - Compile each with the appropriate backend
22//! - Link the results into a unified executable
23//!
24//! See the [`hybrid`] module for more details.
25//!
26//! # Example
27//!
28//! ```ignore
29//! use joule_codegen::{Orchestrator, Backend, CompileOptions};
30//!
31//! let orchestrator = Orchestrator::new();
32//!
33//! // Auto-select backend
34//! let result = orchestrator.compile(&mir_context, CompileOptions::default())?;
35//!
36//! // Or specify backend explicitly
37//! let result = orchestrator.compile_with_backend(
38//!     &mir_context,
39//!     Backend::Llvm,
40//!     CompileOptions::release(),
41//! )?;
42//!
43//! // For hybrid CPU/GPU compilation
44//! use joule_codegen::hybrid::{HybridCompiler, HybridTarget};
45//! let hybrid = HybridCompiler::new(HybridTarget::Cuda);
46//! let result = hybrid.compile_hybrid(&mir_context, 2)?;
47//! ```

60/// Errors that can occur during code generation
61#[derive(Error, Debug)]
62pub enum CodegenError {
63    #[error("Backend '{backend}' is not available: {reason}")]
64    BackendUnavailable { backend: String, reason: String },
65
66    #[error("Compilation failed in {backend} backend: {message}")]
67    CompilationFailed { backend: String, message: String },
68
69    #[error("Backend '{backend}' does not support target '{target}'")]
70    UnsupportedTarget { backend: String, target: String },
71
72    #[error("No suitable backend found for the given code and options")]
73    NoSuitableBackend,
74
75    #[cfg(feature = "mlir")]
76    #[error("MLIR backend error: {0}")]
77    MlirError(#[from] joule_codegen_mlir::MlirError),
78
79    #[cfg(not(feature = "mlir"))]
80    #[error("MLIR backend not available (compile with 'mlir' feature)")]
81    MlirError(String),
82
83    #[error("IO error: {0}")]
84    IoError(#[from] std::io::Error),
85}
86
87pub type CodegenResult<T> = Result<T, CodegenError>;
