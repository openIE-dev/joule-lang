1//! C code generation backend for Joule
2//!
3//! Translates MIR to C source code for the bootstrap self-hosting process.
4//! The generated C code is compiled with a standard C compiler (gcc/clang)
5//! to produce the bootstrap compiler binary.
6
7pub mod emit;
8pub mod linker_script;
9pub mod target;
10mod types;
11
12pub use emit::TargetConfig;
13
14use anyhow::Result;
15use joule_common::Symbol;
16use joule_mir::MirContext;
17use std::collections::{HashMap, HashSet};
18
19/// Compile a MIR context to C source code.
20///
21/// Returns the complete C source as a string, ready to be written to a `.c` file
22/// and compiled with `cc -O2`.
23pub fn compile_to_c(mir: &MirContext) -> Result<String> {
24    compile_to_c_with_modules(mir, &HashMap::new(), &HashSet::new())
25}
26
27/// Compile a MIR context to C source code with module-qualified type names.
28///
29/// - `file_to_module` maps file IDs to module names (e.g., `file_id=5` → "ast")
30/// - `conflicting_types` are type names that need module qualification (e.g., "Function")
31pub fn compile_to_c_with_modules(
32    mir: &MirContext,
33    file_to_module: &HashMap<u32, Symbol>,
34    conflicting_types: &HashSet<Symbol>,
35) -> Result<String> {
36    compile_to_c_with_config(mir, file_to_module, conflicting_types, TargetConfig::default())
37}
38
39/// Compile a MIR context to C source code with target configuration.
40///
41/// When `target_config.freestanding` is true, generates C code that does not
42/// depend on libc (no malloc, stdio). Instead, it declares extern functions
43/// (`jrt_alloc`, `jrt_write_stdout`, `jrt_panic_handler`) that the user provides.
44pub fn compile_to_c_with_config(
45    mir: &MirContext,
46    file_to_module: &HashMap<u32, Symbol>,
47    conflicting_types: &HashSet<Symbol>,
48    target_config: TargetConfig,
49) -> Result<String> {
50    let emitter = emit::CEmitter::new_with_modules_and_config(
51        mir,
52        file_to_module,
53        conflicting_types,
54        target_config,
55    );
56    Ok(emitter.emit())
57}
