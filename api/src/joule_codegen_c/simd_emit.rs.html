1//! Simd[T; N] C code emission.
2//!
3//! Generates C struct typedefs and helper functions for SIMD vector types.
4//! Maps to platform intrinsics where available (x86 SSE/AVX, ARM NEON),
5//! with scalar array fallback for portability.
6
7use crate::types::ty_to_c;
8use joule_mir::{MirContext, Ty};
9use std::collections::HashSet;
10use std::fmt::Write;
11
12/// Type info for a Simd instantiation.
13#[derive(Debug, Clone, PartialEq, Eq, Hash)]
14pub struct SimdTypeInfo {
15    pub elem_c: String,
16    pub elem_ty_c: String,
17    pub lanes: u32,
18}
19
20/// Collect all Simd type instantiations from MIR.
21fn collect_simd_types(mir_ctx: &MirContext) -> Vec<SimdTypeInfo> {
22    let mut types = HashSet::new();
23
24    for (_, func) in &mir_ctx.functions {
25        for local in &func.locals {
26            collect_from_ty(&local.ty, &mut types);
27        }
28    }
29    for s in &mir_ctx.structs {
30        for f in &s.fields {
31            collect_from_ty(&Ty::from_hir(&f.ty), &mut types);
32        }
33    }
34
35    let mut sorted: Vec<_> = types.into_iter().collect();
36    sorted.sort_by(|a, b| a.elem_c.cmp(&b.elem_c).then(a.lanes.cmp(&b.lanes)));
37    sorted
38}
39
40fn collect_from_ty(ty: &Ty, types: &mut HashSet<SimdTypeInfo>) {
41    match ty {
42        Ty::Simd { element, lanes } => {
43            let elem_c = ty_to_c(element).replace('*', "ptr").replace(' ', "_");
44            types.insert(SimdTypeInfo {
45                elem_c,
46                elem_ty_c: ty_to_c(element),
47                lanes: *lanes,
48            });
49            collect_from_ty(element, types);
50        }
51        Ty::Ref { inner, .. } | Ty::RawPtr { inner, .. } => collect_from_ty(inner, types),
52        Ty::Array { element, .. } | Ty::Slice { element, .. } => collect_from_ty(element, types),
53        Ty::Tuple(elts) => {
54            for e in elts {
55                collect_from_ty(e, types);
56            }
57        }
58        Ty::Generic { args, .. } => {
59            for a in args {
60                collect_from_ty(a, types);
61            }
62        }
63        _ => {}
64    }
65}
66
67/// Emit Simd struct typedefs and helper functions.
68pub fn emit_simd_typedefs(out: &mut String, mir_ctx: &MirContext) {
69    let types = collect_simd_types(mir_ctx);
70    if types.is_empty() {
71        return;
72    }
73
74    let _ = writeln!(out, "\n/* ===== SIMD Vector Types ===== */\n");
75
76    // Platform detection header guards
77    let _ = writeln!(out, "#if defined(__x86_64__) || defined(_M_X64) || defined(__i386__) || defined(_M_IX86)");
78    let _ = writeln!(out, "  #include <immintrin.h>");
79    let _ = writeln!(out, "  #define JOULE_SIMD_X86 1");
80    let _ = writeln!(out, "#elif defined(__aarch64__) || defined(_M_ARM64)");
81    let _ = writeln!(out, "  #include <arm_neon.h>");
82    let _ = writeln!(out, "  #define JOULE_SIMD_NEON 1");
83    let _ = writeln!(out, "#endif\n");
84
85    for info in &types {
86        let name = format!("JouleSimd_{}_{}", info.elem_c, info.lanes);
87        let elem = &info.elem_ty_c;
88        let lanes = info.lanes;
89
90        // Struct typedef — portable scalar fallback
91        let _ = writeln!(out, "typedef struct {name} {{");
92        let _ = writeln!(out, "    {elem} data[{lanes}];");
93        let _ = writeln!(out, "}} {name};\n");
94
95        // splat
96        let _ = writeln!(out, "static inline {name} {name}_splat({elem} val) {{");
97        let _ = writeln!(out, "    {name} v;");
98        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) v.data[i] = val;");
99        let _ = writeln!(out, "    return v;");
100        let _ = writeln!(out, "}}\n");
101
102        // from_array
103        let _ = writeln!(out, "static inline {name} {name}_from_array(const {elem}* arr) {{");
104        let _ = writeln!(out, "    {name} v;");
105        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) v.data[i] = arr[i];");
106        let _ = writeln!(out, "    return v;");
107        let _ = writeln!(out, "}}\n");
108
109        // to_array
110        let _ = writeln!(out, "static inline void {name}_to_array({name}* v, {elem}* arr) {{");
111        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) arr[i] = v->data[i];");
112        let _ = writeln!(out, "}}\n");
113
114        // Binary ops: add, sub, mul, div
115        for op_name in &["add", "sub", "mul", "div"] {
116            let c_op = match *op_name {
117                "add" => "+",
118                "sub" => "-",
119                "mul" => "*",
120                "div" => "/",
121                _ => "+",
122            };
123            let _ = writeln!(out, "static inline {name} {name}_{op_name}({name}* a, {name}* b) {{");
124            let _ = writeln!(out, "    {name} r;");
125            let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) r.data[i] = a->data[i] {c_op} b->data[i];");
126            let _ = writeln!(out, "    return r;");
127            let _ = writeln!(out, "}}\n");
128        }
129
130        // min/max
131        let _ = writeln!(out, "static inline {name} {name}_min({name}* a, {name}* b) {{");
132        let _ = writeln!(out, "    {name} r;");
133        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) r.data[i] = a->data[i] < b->data[i] ? a->data[i] : b->data[i];");
134        let _ = writeln!(out, "    return r;");
135        let _ = writeln!(out, "}}\n");
136
137        let _ = writeln!(out, "static inline {name} {name}_max({name}* a, {name}* b) {{");
138        let _ = writeln!(out, "    {name} r;");
139        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) r.data[i] = a->data[i] > b->data[i] ? a->data[i] : b->data[i];");
140        let _ = writeln!(out, "    return r;");
141        let _ = writeln!(out, "}}\n");
142
143        // abs
144        let _ = writeln!(out, "static inline {name} {name}_abs({name}* v) {{");
145        let _ = writeln!(out, "    {name} r;");
146        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) r.data[i] = v->data[i] < 0 ? -v->data[i] : v->data[i];");
147        let _ = writeln!(out, "    return r;");
148        let _ = writeln!(out, "}}\n");
149
150        // neg
151        let _ = writeln!(out, "static inline {name} {name}_neg({name}* v) {{");
152        let _ = writeln!(out, "    {name} r;");
153        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) r.data[i] = -v->data[i];");
154        let _ = writeln!(out, "    return r;");
155        let _ = writeln!(out, "}}\n");
156
157        // sum (horizontal reduction)
158        let _ = writeln!(out, "static inline {elem} {name}_sum({name}* v) {{");
159        let _ = writeln!(out, "    {elem} acc = 0;");
160        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) acc += v->data[i];");
161        let _ = writeln!(out, "    return acc;");
162        let _ = writeln!(out, "}}\n");
163
164        // load from pointer + offset
165        let _ = writeln!(out, "static inline {name} {name}_load(const {elem}* ptr, usize offset) {{");
166        let _ = writeln!(out, "    {name} v;");
167        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) v.data[i] = ptr[offset + i];");
168        let _ = writeln!(out, "    return v;");
169        let _ = writeln!(out, "}}\n");
170
171        // store to pointer + offset
172        let _ = writeln!(out, "static inline void {name}_store({name}* v, {elem}* ptr, usize offset) {{");
173        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) ptr[offset + i] = v->data[i];");
174        let _ = writeln!(out, "}}\n");
175
176        // eq (element-wise comparison → bool)
177        let _ = writeln!(out, "static inline bool {name}_eq({name}* a, {name}* b) {{");
178        let _ = writeln!(out, "    for (usize i = 0; i < {lanes}; i++) if (a->data[i] != b->data[i]) return false;");
179        let _ = writeln!(out, "    return true;");
180        let _ = writeln!(out, "}}\n");
181    }
182}
