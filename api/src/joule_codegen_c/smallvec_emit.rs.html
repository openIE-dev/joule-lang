1//! SmallVec[T; N] C code emission.
2//!
3//! Generates C struct typedefs and helper functions for SmallVec types.
4//! SmallVec stores up to N elements inline, then spills to a heap allocation.
5
6use crate::types::ty_to_c;
7use joule_mir::{MirContext, Ty};
8use std::collections::HashSet;
9use std::fmt::Write;
10
11/// Type info for a SmallVec instantiation.
12#[derive(Debug, Clone, PartialEq, Eq, Hash)]
13pub struct SmallVecTypeInfo {
14    pub elem_c: String,
15    pub elem_ty_c: String,
16    pub capacity: u32,
17}
18
19/// Collect all SmallVec type instantiations from MIR.
20fn collect_smallvec_types(mir_ctx: &MirContext) -> Vec<SmallVecTypeInfo> {
21    let mut types = HashSet::new();
22
23    for (_, func) in &mir_ctx.functions {
24        for local in &func.locals {
25            collect_from_ty(&local.ty, &mut types);
26        }
27    }
28    for s in &mir_ctx.structs {
29        for f in &s.fields {
30            collect_from_ty(&Ty::from_hir(&f.ty), &mut types);
31        }
32    }
33
34    let mut sorted: Vec<_> = types.into_iter().collect();
35    sorted.sort_by(|a, b| a.elem_c.cmp(&b.elem_c).then(a.capacity.cmp(&b.capacity)));
36    sorted
37}
38
39fn collect_from_ty(ty: &Ty, types: &mut HashSet<SmallVecTypeInfo>) {
40    match ty {
41        Ty::SmallVec { element, capacity } => {
42            let elem_c = ty_to_c(element).replace('*', "ptr").replace(' ', "_");
43            types.insert(SmallVecTypeInfo {
44                elem_c,
45                elem_ty_c: ty_to_c(element),
46                capacity: *capacity,
47            });
48            collect_from_ty(element, types);
49        }
50        Ty::Ref { inner, .. } | Ty::RawPtr { inner, .. } => collect_from_ty(inner, types),
51        Ty::Array { element, .. } | Ty::Slice { element, .. } => collect_from_ty(element, types),
52        Ty::Tuple(elts) => {
53            for e in elts {
54                collect_from_ty(e, types);
55            }
56        }
57        Ty::Generic { args, .. } => {
58            for a in args {
59                collect_from_ty(a, types);
60            }
61        }
62        _ => {}
63    }
64}
65
66/// Emit SmallVec struct typedefs and helper functions.
67pub fn emit_smallvec_typedefs(out: &mut String, mir_ctx: &MirContext) {
68    let types = collect_smallvec_types(mir_ctx);
69    if types.is_empty() {
70        return;
71    }
72
73    let _ = writeln!(out, "\n/* ===== SmallVec Types ===== */\n");
74
75    for info in &types {
76        let name = format!("JouleSmallVec_{}_{}", info.elem_c, info.capacity);
77        let elem = &info.elem_ty_c;
78        let cap = info.capacity;
79
80        // Struct typedef
81        let _ = writeln!(out, "typedef struct {name} {{");
82        let _ = writeln!(out, "    {elem} inline_buf[{cap}];");
83        let _ = writeln!(out, "    {elem}* heap;");
84        let _ = writeln!(out, "    usize len;");
85        let _ = writeln!(out, "    usize heap_capacity;");
86        let _ = writeln!(out, "    u8 spilled;");
87        let _ = writeln!(out, "}} {name};\n");
88
89        // Constructor
90        let _ = writeln!(out, "static inline {name} {name}_new(void) {{");
91        let _ = writeln!(out, "    {name} sv;");
92        let _ = writeln!(out, "    sv.heap = (void*)0;");
93        let _ = writeln!(out, "    sv.len = 0;");
94        let _ = writeln!(out, "    sv.heap_capacity = 0;");
95        let _ = writeln!(out, "    sv.spilled = 0;");
96        let _ = writeln!(out, "    return sv;");
97        let _ = writeln!(out, "}}\n");
98
99        // data() — returns pointer to active buffer
100        let _ = writeln!(out, "static inline {elem}* {name}_data({name}* sv) {{");
101        let _ = writeln!(out, "    return sv->spilled ? sv->heap : sv->inline_buf;");
102        let _ = writeln!(out, "}}\n");
103
104        // push
105        let _ = writeln!(out, "static inline void {name}_push({name}* sv, {elem} val) {{");
106        let _ = writeln!(out, "    if (!sv->spilled && sv->len < {cap}) {{");
107        let _ = writeln!(out, "        sv->inline_buf[sv->len++] = val;");
108        let _ = writeln!(out, "        return;");
109        let _ = writeln!(out, "    }}");
110        let _ = writeln!(out, "    if (!sv->spilled) {{");
111        let _ = writeln!(out, "        usize new_cap = {cap} * 2;");
112        let _ = writeln!(out, "        sv->heap = ({elem}*)jrt_alloc(new_cap * sizeof({elem}));");
113        let _ = writeln!(out, "        for (usize i = 0; i < sv->len; i++) sv->heap[i] = sv->inline_buf[i];");
114        let _ = writeln!(out, "        sv->heap_capacity = new_cap;");
115        let _ = writeln!(out, "        sv->spilled = 1;");
116        let _ = writeln!(out, "    }}");
117        let _ = writeln!(out, "    if (sv->len >= sv->heap_capacity) {{");
118        let _ = writeln!(out, "        usize new_cap = sv->heap_capacity * 2;");
119        let _ = writeln!(out, "        {elem}* new_buf = ({elem}*)jrt_alloc(new_cap * sizeof({elem}));");
120        let _ = writeln!(out, "        for (usize i = 0; i < sv->len; i++) new_buf[i] = sv->heap[i];");
121        let _ = writeln!(out, "        jrt_dealloc(sv->heap);");
122        let _ = writeln!(out, "        sv->heap = new_buf;");
123        let _ = writeln!(out, "        sv->heap_capacity = new_cap;");
124        let _ = writeln!(out, "    }}");
125        let _ = writeln!(out, "    sv->heap[sv->len++] = val;");
126        let _ = writeln!(out, "}}\n");
127
128        // pop
129        let _ = writeln!(out, "static inline {elem} {name}_pop({name}* sv) {{");
130        let _ = writeln!(out, "    {elem}* data = {name}_data(sv);");
131        let _ = writeln!(out, "    return data[--sv->len];");
132        let _ = writeln!(out, "}}\n");
133
134        // get
135        let _ = writeln!(out, "static inline {elem} {name}_get({name}* sv, usize idx) {{");
136        let _ = writeln!(out, "    {elem}* data = {name}_data(sv);");
137        let _ = writeln!(out, "    return data[idx];");
138        let _ = writeln!(out, "}}\n");
139
140        // set
141        let _ = writeln!(out, "static inline void {name}_set({name}* sv, usize idx, {elem} val) {{");
142        let _ = writeln!(out, "    {elem}* data = {name}_data(sv);");
143        let _ = writeln!(out, "    data[idx] = val;");
144        let _ = writeln!(out, "}}\n");
145
146        // len
147        let _ = writeln!(out, "static inline usize {name}_len({name}* sv) {{");
148        let _ = writeln!(out, "    return sv->len;");
149        let _ = writeln!(out, "}}\n");
150
151        // is_empty
152        let _ = writeln!(out, "static inline bool {name}_is_empty({name}* sv) {{");
153        let _ = writeln!(out, "    return sv->len == 0;");
154        let _ = writeln!(out, "}}\n");
155
156        // capacity
157        let _ = writeln!(out, "static inline usize {name}_capacity({name}* sv) {{");
158        let _ = writeln!(out, "    return sv->spilled ? sv->heap_capacity : {cap};");
159        let _ = writeln!(out, "}}\n");
160
161        // spilled
162        let _ = writeln!(out, "static inline bool {name}_spilled({name}* sv) {{");
163        let _ = writeln!(out, "    return sv->spilled;");
164        let _ = writeln!(out, "}}\n");
165
166        // clear
167        let _ = writeln!(out, "static inline void {name}_clear({name}* sv) {{");
168        let _ = writeln!(out, "    sv->len = 0;");
169        let _ = writeln!(out, "}}\n");
170
171        // drop (free heap if spilled)
172        let _ = writeln!(out, "static inline void {name}_drop({name}* sv) {{");
173        let _ = writeln!(out, "    if (sv->spilled && sv->heap) jrt_dealloc(sv->heap);");
174        let _ = writeln!(out, "    sv->heap = (void*)0;");
175        let _ = writeln!(out, "    sv->len = 0;");
176        let _ = writeln!(out, "    sv->spilled = 0;");
177        let _ = writeln!(out, "}}\n");
178    }
179}
