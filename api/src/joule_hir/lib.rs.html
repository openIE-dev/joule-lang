1//! High-level Intermediate Representation (HIR) for Joule
2//!
3//! HIR is a typed, desugared representation of the AST. It includes:
4//! - Resolved types for all expressions
5//! - Resolved names (symbols point to definitions)
6//! - Desugared syntax (no syntactic sugar)
7//! - Type annotations on all nodes
8//! - Energy budget metadata from `#[energy_budget(...)]` attributes
9

47/// Parsed function attributes for code generation decisions
48#[derive(Debug, Clone, Default)]
49pub struct FunctionAttributes {
50    /// Function is a GPU kernel (marked with `#[gpu_kernel]` or `#[gpu]`)
51    pub is_gpu_kernel: bool,
52    /// Function should have thermal-aware code generation (`#[thermal_aware]`)
53    pub is_thermal_aware: bool,
54    /// Function targets a specific processor (`#[target(cpu|gpu|tpu|npu)]`)
55    pub target: Option<ProcessorTarget>,
56    /// Function should be inlined (`#[inline]` or `#[inline(always)]`)
57    pub inline: InlineHint,
58    /// Function is marked as cold/unlikely to be called (`#[cold]`)
59    pub is_cold: bool,
60    /// Function is a test (`#[test]`) — included only in `--test` mode
61    pub is_test: bool,
62    /// Function is a benchmark (`#[bench]`) — included only in `--bench` mode
63    pub is_bench: bool,
64    /// Function should be ignored in test/bench mode (`#[ignore]`)
65    pub is_ignored: bool,
66    /// Raw attributes for extensions
67    pub raw: Vec<(String, Vec<(String, String)>)>,
68}

277    Channel { element_ty: Box<Self> },

347/// Pattern (simplified from AST)

349pub enum Pattern {
350    /// Identifier pattern
351    Ident {
352        id: HirId,
353        name: Symbol,
354        mutable: bool,

356    },
357    /// Wildcard pattern
358    Wildcard { id: HirId, ty: Ty },
359    /// Tuple pattern
360    Tuple {
361        id: HirId,

363        ty: Ty,
364    },
365    /// Struct pattern
366    Struct {
367        id: HirId,
368        def_id: HirId,

370        ty: Ty,

382    /// Or-pattern (e.g., `1 | 2 | 3`)
383    Or {
384        id: HirId,

386        ty: Ty,

700impl Ty {
701    /// Check if this type is an integer type
702    pub const fn is_int(&self) -> bool {
703        matches!(self, Self::Int(_) | Self::Uint(_))
704    }
705
706    /// Check if this type is a float type
707    pub const fn is_float(&self) -> bool {
708        matches!(self, Self::Float(_))
709    }
710
711    /// Check if this type is numeric
712    pub fn is_numeric(&self) -> bool {
713        self.is_int() || self.is_float()
714    }
715
716    /// Check if this type is copyable
717    pub fn is_copy(&self) -> bool {
718        match self {
719            Self::Int(_)
720            | Self::Uint(_)
721            | Self::Float(_)
722            | Self::Bool
723            | Self::Char
724            | Self::Unit => true,
725            Self::Ref { .. } => true,
726            Self::Tuple(tys) => tys.iter().all(Ty::is_copy),
727            _ => false,
728        }
729    }
730
731    /// Get the default value for this type (for uninitialized variables)
732    pub const fn default_value(&self) -> Option<Literal> {
733        match self {
734            Self::Int(int_ty) => Some(Literal::Int(0, *int_ty)),
735            Self::Uint(uint_ty) => Some(Literal::Uint(0, *uint_ty)),
736            Self::Float(float_ty) => Some(Literal::Float(0.0, *float_ty)),
737            Self::Bool => Some(Literal::Bool(false)),
738            Self::Char => Some(Literal::Char('\0')),
739            _ => None,
740        }
741    }
742}

744impl Pattern {
745    /// Get the type of this pattern
746    pub const fn ty(&self) -> &Ty {
747        match self {
748            Self::Ident { ty, .. } => ty,
749            Self::Wildcard { ty, .. } => ty,
750            Self::Tuple { ty, .. } => ty,
751            Self::Struct { ty, .. } => ty,
752            Self::Variant { ty, .. } => ty,
753            Self::Literal { ty, .. } => ty,
754            Self::Or { ty, .. } => ty,
755            Self::Range { ty, .. } => ty,
756        }
757    }
758}
