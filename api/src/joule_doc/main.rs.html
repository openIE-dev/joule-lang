125/// Documented item
126#[derive(Debug, Clone, Serialize, Deserialize)]
127struct ItemDoc {
128    kind: ItemKind,
129    name: String,
130    visibility: String,
131    signature: String,
132    doc_comment: Option<String>,
133    /// Energy annotation
134    energy: Option<EnergyAnnotation>,
135    /// Source location
136    source: SourceLoc,
137    /// Examples extracted from doc comments
138    examples: Vec<Example>,
139    /// Related items
140    see_also: Vec<String>,
141}

143#[derive(Debug, Clone, Serialize, Deserialize)]
144enum ItemKind {
145    Function,
146    Struct,
147    Enum,
148    Trait,
149    Impl,
150    Const,
151    Static,
152    Type,
153    Macro,
154    Task,    // Joule-specific: task fn
155    Channel, // Joule-specific: channel type
156}
157
158impl std::fmt::Display for ItemKind {
159    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
160        match self {
161            Self::Function => write!(f, "fn"),
162            Self::Struct => write!(f, "struct"),
163            Self::Enum => write!(f, "enum"),
164            Self::Trait => write!(f, "trait"),
165            Self::Impl => write!(f, "impl"),
166            Self::Const => write!(f, "const"),
167            Self::Static => write!(f, "static"),
168            Self::Type => write!(f, "type"),
169            Self::Macro => write!(f, "macro"),
170            Self::Task => write!(f, "task"),
171            Self::Channel => write!(f, "channel"),
172        }
173    }
174}

258    /// Generate documentation for a package
259    fn generate(&self) -> Result<PackageDoc> {
260        info!("Generating documentation from {:?}", self.config.source_dir);
261
262        // Read package info
263        let cargo_toml = self.config.source_dir.parent().unwrap().join("Joule.toml");
264        let (name, version, description) = if cargo_toml.exists() {
265            let content = fs::read_to_string(&cargo_toml)?;
266            let config: toml::Value = toml::from_str(&content)?;
267            (
268                config
269                    .get("package")
270                    .and_then(|p| p.get("name"))
271                    .and_then(|n| n.as_str())
272                    .unwrap_or("unnamed")
273                    .to_string(),
274                config
275                    .get("package")
276                    .and_then(|p| p.get("version"))
277                    .and_then(|v| v.as_str())
278                    .unwrap_or("0.0.0")
279                    .to_string(),
280                config
281                    .get("package")
282                    .and_then(|p| p.get("description"))
283                    .and_then(|d| d.as_str())
284                    .map(String::from),
285            )
286        } else {
287            ("unnamed".to_string(), "0.0.0".to_string(), None)
288        };
289
290        // Parse all source files
291        let mut modules = Vec::new();
292        for entry in WalkDir::new(&self.config.source_dir) {
293            let entry = entry?;
294            if entry.path().extension().is_some_and(|ext| ext == "joule") {
295                if let Ok(module) = self.parse_module(entry.path()) {
296                    modules.push(module);
297                }
298            }
299        }
300
301        // Compute energy profile
302        let energy_profile = if self.config.include_energy {
303            Some(self.compute_energy_profile(&modules))
304        } else {
305            None
306        };
307
308        let package = PackageDoc {
309            name,
310            version,
311            description,
312            modules,
313            energy_profile,
314        };
315
316        // Write output
317        self.write_output(&package)?;
318
319        Ok(package)
320    }

322    /// Parse a single module
323    fn parse_module(&self, path: &Path) -> Result<ModuleDoc> {
324        let content = fs::read_to_string(path)?;
325        let relative = path
326            .strip_prefix(&self.config.source_dir)
327            .unwrap_or(path)
328            .to_string_lossy()
329            .to_string();
330
331        let name = path
332            .file_stem()
333            .map(|s| s.to_string_lossy().to_string())
334            .unwrap_or_else(|| "unknown".to_string());
335
336        // Extract doc comments and items
337        let doc_comment = self.extract_module_doc(&content);
338        let items = self.extract_items(&content, &relative);
339        let submodules = self.find_submodules(path);
340
341        Ok(ModuleDoc {
342            name,
343            path: relative,
344            doc_comment,
345            items,
346            submodules,
347        })
348    }

350    /// Extract module-level doc comment
351    fn extract_module_doc(&self, content: &str) -> Option<String> {
352        let mut doc_lines = Vec::new();
353        for line in content.lines() {
354            if line.starts_with("//!") {
355                doc_lines.push(line.trim_start_matches("//!").trim());
356            } else if !line.trim().is_empty() {
357                break;
358            }
359        }
360
361        if doc_lines.is_empty() {
362            None
363        } else {
364            Some(doc_lines.join("\n"))
365        }
366    }

368    /// Extract documented items from source
369    fn extract_items(&self, content: &str, file: &str) -> Vec<ItemDoc> {
370        let mut items = Vec::new();
371        let mut current_doc = Vec::new();
372        let mut line_number = 0u32;
373
374        for line in content.lines() {
375            line_number += 1;
376
377            // Collect doc comments
378            if line.trim().starts_with("///") {
379                current_doc.push(line.trim().trim_start_matches("///").trim().to_string());
380                continue;
381            }
382
383            // Check for item declarations
384            let trimmed = line.trim();
385
386            // Function
387            if let Some(rest) = trimmed
388                .strip_prefix("fn ")
389                .or(trimmed.strip_prefix("pub fn "))
390            {
391                if let Some(item) = self.parse_function(rest, &current_doc, file, line_number) {
392                    items.push(item);
393                }
394            }
395            // Task function
396            else if let Some(rest) = trimmed
397                .strip_prefix("task fn ")
398                .or(trimmed.strip_prefix("pub task fn "))
399            {
400                if let Some(item) = self.parse_task(rest, &current_doc, file, line_number) {
401                    items.push(item);
402                }
403            }
404            // Struct
405            else if let Some(rest) = trimmed
406                .strip_prefix("struct ")
407                .or(trimmed.strip_prefix("pub struct "))
408            {
409                if let Some(item) = self.parse_struct(rest, &current_doc, file, line_number) {
410                    items.push(item);
411                }
412            }
413            // Enum
414            else if let Some(rest) = trimmed
415                .strip_prefix("enum ")
416                .or(trimmed.strip_prefix("pub enum "))
417            {
418                if let Some(item) = self.parse_enum(rest, &current_doc, file, line_number) {
419                    items.push(item);
420                }
421            }
422            // Trait
423            else if let Some(rest) = trimmed
424                .strip_prefix("trait ")
425                .or(trimmed.strip_prefix("pub trait "))
426            {
427                if let Some(item) = self.parse_trait(rest, &current_doc, file, line_number) {
428                    items.push(item);
429                }
430            }
431            // Const
432            else if let Some(rest) = trimmed
433                .strip_prefix("const ")
434                .or(trimmed.strip_prefix("pub const "))
435            {
436                if let Some(item) = self.parse_const(rest, &current_doc, file, line_number) {
437                    items.push(item);
438                }
439            }
440
441            // Clear doc comment if not followed by item
442            if !trimmed.is_empty() && !trimmed.starts_with("//") {
443                current_doc.clear();
444            }
445        }
446
447        if !self.config.document_private {
448            items.retain(|item| item.visibility == "pub");
449        }
450        items
451    }

495    fn create_item(
496        &self,
497        kind: ItemKind,
498        name: String,
499        signature: String,
500        doc: &[String],
501        file: &str,
502        line: u32,

517                column: 1,
518            },
519            examples,
520            see_also,
521        }
522    }

524    /// Parse doc comment to extract examples, energy info, etc.
525    fn parse_doc_comment(
526        &self,
527        doc: &str,
528    ) -> (
529        Option<String>,
530        Vec<Example>,
531        Option<EnergyAnnotation>,
532        Vec<String>,
533    ) {
534        let mut main_doc = Vec::new();
535        let mut examples = Vec::new();
536        let mut energy = None;
537        let mut see_also = Vec::new();
538
539        let mut in_example = false;
540        let mut example_code = Vec::new();
541        let mut example_desc = None;
542
543        for line in doc.lines() {
544            // Check for example block
545            if line.contains("```joule") || line.contains("```") && line.contains("example") {
546                in_example = true;
547                continue;
548            }
549            if in_example && line.contains("```") {
550                examples.push(Example {
551                    code: example_code.join("\n"),
552                    description: example_desc.take(),
553                    energy_cost: None,
554                });
555                example_code.clear();
556                in_example = false;
557                continue;
558            }
559            if in_example {
560                example_code.push(line.to_string());
561                continue;
562            }
563
564            // Check for energy annotation
565            if line.starts_with("# Energy:") || line.starts_with("Energy:") {
566                let cost_str = line.split(':').nth(1).unwrap_or("0").trim();
567                if let Ok(cost) = cost_str.parse::<f64>() {
568                    energy = Some(EnergyAnnotation {
569                        base_cost: cost,
570                        complexity: "O(1)".to_string(),
571                        formula: None,
572                        hardware_specific: HashMap::new(),
573                    });
574                }
575                continue;
576            }
577
578            // Check for complexity
579            if line.starts_with("# Complexity:") {
580                if let Some(ref mut e) = energy {
581                    e.complexity = line.split(':').nth(1).unwrap_or("O(1)").trim().to_string();
582                }
583                continue;
584            }
585
586            // Check for see also
587            if line.starts_with("# See also:") {
588                let refs: Vec<String> = line
589                    .split(':')
590                    .nth(1)
591                    .unwrap_or("")
592                    .split(',')
593                    .map(|s| s.trim().to_string())
594                    .filter(|s| !s.is_empty())
595                    .collect();
596                see_also.extend(refs);
597                continue;
598            }
599
600            // Check for example description
601            if line.starts_with("# Example") {
602                example_desc = line.split(':').nth(1).map(|s| s.trim().to_string());
603                continue;
604            }
605
606            main_doc.push(line.to_string());
607        }
608
609        let doc_text = main_doc.join("\n");
610        let doc_comment = if doc_text.is_empty() {
611            None
612        } else {
613            Some(doc_text)
614        };
615
616        (doc_comment, examples, energy, see_also)
617    }

619    /// Find submodules
620    fn find_submodules(&self, path: &Path) -> Vec<String> {
621        let dir = path.parent().unwrap().join(path.file_stem().unwrap());
622        if dir.is_dir() {
623            fs::read_dir(dir)
624                .map(|entries| {
625                    entries
626                        .filter_map(|e| e.ok())
627                        .filter(|e| e.path().extension().is_some_and(|ext| ext == "joule"))
628                        .filter_map(|e| {
629                            e.path()
630                                .file_stem()
631                                .map(|s| s.to_string_lossy().to_string())
632                        })
633                        .collect()
634                })
635                .unwrap_or_default()
636        } else {
637            Vec::new()
638        }
639    }

641    /// Compute energy profile for a package
642    fn compute_energy_profile(&self, modules: &[ModuleDoc]) -> EnergyProfile {
643        let mut total_cost = 0.0;
644        let mut hotspots = Vec::new();
645
646        for module in modules {
647            for item in &module.items {
648                if let Some(ref energy) = item.energy {
649                    total_cost += energy.base_cost;
650
651                    if energy.base_cost > 100.0 {
652                        hotspots.push(Hotspot {
653                            item: format!("{}::{}", module.name, item.name),
654                            cost: energy.base_cost,
655                            suggestion: Some("Consider optimizing this function".to_string()),
656                        });
657                    }
658                }
659            }
660        }
661
662        // Sort hotspots by cost
663        hotspots.sort_by(|a, b| {
664            b.cost
665                .partial_cmp(&a.cost)
666                .unwrap_or(std::cmp::Ordering::Equal)
667        });
668
669        let recommendations = if hotspots.is_empty() {
670            vec!["No significant energy hotspots found.".to_string()]
671        } else {
672            hotspots
673                .iter()
674                .take(3)
675                .map(|h| format!("Optimize '{}' - estimated {:.2} µJ", h.item, h.cost))
676                .collect()
677        };
678
679        EnergyProfile {
680            total_estimated_cost: total_cost,
681            hotspots,
682            recommendations,
683        }
684    }

697    fn render_markdown(&self, md: &str) -> String {
698        let opts = Options::all();
699        let p = MdParser::new_ext(md, opts);
700        let mut out = String::new();
701        html::push_html(&mut out, p);
702        out

705    fn write_html(&self, package: &PackageDoc) -> Result<()> {
706        // Write index
707        let index_html = self.handlebars.render("index", package)?;
708        fs::write(self.config.output_dir.join("index.html"), index_html)?;
709
710        // Write module pages
711        for module in &package.modules {
712            let mut rendered_module = module.clone();
713            if let Some(ref doc) = rendered_module.doc_comment {
714                rendered_module.doc_comment = Some(self.render_markdown(doc));
715            }
716            let module_html = self.handlebars.render("module", &rendered_module)?;
717            let module_path = self.config.output_dir.join(format!("{}.html", module.name));
718            fs::write(module_path, module_html)?;
719        }
720
721        // Write CSS
722        fs::write(self.config.output_dir.join("style.css"), STYLE_CSS)?;
723
724        info!("HTML documentation written to {:?}", self.config.output_dir);
725        Ok(())
726    }

728    fn write_markdown(&self, package: &PackageDoc) -> Result<()> {
729        let mut content = String::new();
730
731        // Package header
732        content.push_str(&format!("# {}\n\n", package.name));
733        if let Some(ref desc) = package.description {
734            content.push_str(&format!("{}\n\n", desc));
735        }
736
737        // Modules
738        content.push_str("## Modules\n\n");
739        for module in &package.modules {
740            content.push_str(&format!("### {}\n\n", module.name));
741            if let Some(ref doc) = module.doc_comment {
742                content.push_str(&format!("{}\n\n", doc));
743            }
744
745            for item in &module.items {
746                content.push_str(&format!("#### `{}`\n\n", item.signature));
747                if let Some(ref doc) = item.doc_comment {
748                    content.push_str(&format!("{}\n\n", doc));
749                }
750                if let Some(ref energy) = item.energy {
751                    content.push_str(&format!(
752                        "⚡ **Energy:** {:.2} µJ ({})\n\n",
753                        energy.base_cost, energy.complexity
754                    ));
755                }
756            }
757        }
758
759        // Energy profile
760        if let Some(ref profile) = package.energy_profile {
761            content.push_str("## Energy Profile\n\n");
762            content.push_str(&format!(
763                "**Total estimated energy:** {:.2} µJ\n\n",
764                profile.total_estimated_cost
765            ));

782    }

795
796fn check_coverage(source: &Path) -> Result<()> {
797    let mut total_items = 0;
798    let mut documented_items = 0;
799    let mut undocumented = Vec::new();
800
801    for entry in WalkDir::new(source) {
802        let entry = entry?;
803        if entry.path().extension().is_some_and(|ext| ext == "joule") {
804            let content = fs::read_to_string(entry.path())?;
805            let relative = entry.path().to_string_lossy();
806
807            let mut has_doc = false;
808            for line in content.lines() {
809                if line.trim().starts_with("///") {
810                    has_doc = true;
811                    continue;
812                }
813
814                let trimmed = line.trim();
815                if trimmed.starts_with("pub fn ")
816                    || trimmed.starts_with("pub struct ")
817                    || trimmed.starts_with("pub enum ")
818                    || trimmed.starts_with("pub trait ")
819                {
820                    total_items += 1;
821                    if has_doc {
822                        documented_items += 1;
823                    } else {
824                        // Extract item name
825                        let name = trimmed.split_whitespace().nth(2).unwrap_or("unknown");
826                        undocumented.push(format!("{}::{}", relative, name));
827                    }
828                    has_doc = false;
829                }
830
831                if !trimmed.is_empty() && !trimmed.starts_with("//") {
832                    has_doc = false;
833                }
834            }
835        }
836    }
837
838    let coverage = if total_items > 0 {
839        (documented_items as f64 / total_items as f64) * 100.0
840    } else {
841        100.0
842    };
843
844    println!("Documentation Coverage Report");
845    println!("=============================");
846    println!("Total public items: {}", total_items);
847    println!("Documented items:   {}", documented_items);
848    println!("Coverage:           {:.1}%", coverage);
849
850    if !undocumented.is_empty() {
851        println!("\nUndocumented items:");
852        for item in undocumented.iter().take(20) {
853            println!("  - {}", item);
854        }
855        if undocumented.len() > 20 {
856            println!("  ... and {} more", undocumented.len() - 20);
857        }
858    }
859
860    Ok(())
861}

867/// A single doc test extracted from source
868#[derive(Debug)]
869struct DocTest {
870    /// Source file
871    file: PathBuf,
872    /// Line number where the doc test starts
873    line: u32,
874    /// Item name the doc test belongs to
875    item: String,
876    /// The code to compile/run
877    code: String,
878    /// Expected output (if any)
879    expected_output: Option<String>,
880    /// Flags parsed from the code fence
881    flags: DocTestFlags,
882}
883
884/// Flags that can be set on a doc test
885#[derive(Debug, Default)]
886struct DocTestFlags {
887    /// Don't run this test
888    ignore: bool,
889    /// Don't compile this test (just show it)
890    no_compile: bool,
891    /// Don't run this test (just compile)
892    no_run: bool,
893    /// This test should fail to compile
894    compile_fail: bool,
895    /// This test should panic
896    should_panic: bool,
897}
898
899/// Result of running a doc test
900#[derive(Debug)]
901struct DocTestResult {
902    test: DocTest,
903    status: DocTestStatus,
904    output: Option<String>,
905    error: Option<String>,
906}
907
908#[derive(Debug, Clone, Copy, PartialEq, Eq)]
909enum DocTestStatus {
910    Passed,
911    Failed,
912    Ignored,
913    CompileError,
914}
915
916impl std::fmt::Display for DocTestStatus {
917    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
918        match self {
919            Self::Passed => write!(f, "ok"),
920            Self::Failed => write!(f, "FAILED"),
921            Self::Ignored => write!(f, "ignored"),
922            Self::CompileError => write!(f, "compile error"),
923        }
924    }
925}

927/// Run documentation tests
928fn run_doc_tests(source: &Path, no_run: bool, verbose: bool, filter: Option<&str>) -> Result<()> {
929    println!("Running documentation tests\n");
930
931    // Extract all doc tests
932    let mut doc_tests = Vec::new();
933    extract_doc_tests_from_dir(source, &mut doc_tests)?;
934
935    // Apply filter if specified
936    let doc_tests: Vec<_> = if let Some(pattern) = filter {
937        doc_tests
938            .into_iter()
939            .filter(|t| t.item.contains(pattern) || t.file.to_string_lossy().contains(pattern))
940            .collect()
941    } else {
942        doc_tests
943    };
944
945    if doc_tests.is_empty() {
946        println!("No doc tests found.");
947        return Ok(());
948    }
949
950    println!("Found {} doc tests\n", doc_tests.len());
951
952    // Run each test
953    let mut passed = 0;
954    let mut failed = 0;
955    let mut ignored = 0;
956    let mut results = Vec::new();
957
958    for test in doc_tests {
959        let test_name = format!(
960            "{}::{} (line {})",
961            test.file.file_stem().unwrap_or_default().to_string_lossy(),
962            test.item,
963            test.line
964        );
965
966        if test.flags.ignore || test.flags.no_compile {
967            if verbose {
968                println!("test {} ... ignored", test_name);
969            }
970            ignored += 1;
971            results.push(DocTestResult {
972                test,
973                status: DocTestStatus::Ignored,
974                output: None,
975                error: None,
976            });
977            continue;
978        }
979
980        let result = run_single_doc_test(&test, no_run || test.flags.no_run, verbose);
981
982        match result.status {
983            DocTestStatus::Passed => {
984                passed += 1;
985                if verbose {
986                    println!("test {} ... ok", test_name);
987                } else {
988                    print!(".");
989                }
990            }
991            DocTestStatus::Failed | DocTestStatus::CompileError => {
992                failed += 1;
993                if verbose {
994                    println!("test {} ... FAILED", test_name);
995                    if let Some(ref err) = result.error {
996                        println!("  error: {}", err);
997                    }
998                } else {
999                    print!("F");
1000                }
1001            }
1002            DocTestStatus::Ignored => {
1003                ignored += 1;
1004                if !verbose {
1005                    print!("i");
1006                }
1007            }
1008        }
1009
1010        results.push(result);
1011    }
1012
1013    if !verbose {
1014        println!();
1015    }
1016
1017    println!(
1018        "\nDoc test results: {} passed, {} failed, {} ignored",
1019        passed, failed, ignored
1020    );
1021
1022    // Print failures
1023    let failures: Vec<_> = results
1024        .iter()
1025        .filter(|r| r.status == DocTestStatus::Failed || r.status == DocTestStatus::CompileError)
1026        .collect();
1027
1028    if !failures.is_empty() {
1029        println!("\n---- failures ----\n");
1030        for failure in &failures {
1031            println!(
1032                "{}::{}:{}",
1033                failure.test.file.display(),
1034                failure.test.item,
1035                failure.test.line
1036            );
1037            println!("```");
1038            for (i, line) in failure.test.code.lines().enumerate() {
1039                println!("{:3} | {}", i + 1, line);
1040            }
1041            println!("```");
1042            if let Some(ref output) = failure.output {
1043                println!("Output: {}", output);
1044            }
1045            if let Some(ref err) = failure.error {
1046                println!("Error: {}", err);
1047            }
1048            println!();
1049        }
1050    }
1051
1052    if failed > 0 {
1053        Err(anyhow!("{} doc test(s) failed", failed))
1054    } else {
1055        Ok(())
1056    }
1057}

1059/// Extract doc tests from a directory recursively
1060fn extract_doc_tests_from_dir(dir: &Path, tests: &mut Vec<DocTest>) -> Result<()> {
1061    for entry in WalkDir::new(dir) {
1062        let entry = entry?;
1063        let path = entry.path();
1064
1065        if path
1066            .extension()
1067            .is_some_and(|ext| ext == "joule" || ext == "rs")
1068        {
1069            extract_doc_tests_from_file(path, tests)?;
1070        }
1071    }
1072    Ok(())
1073}

1075/// Extract doc tests from a single file
1076fn extract_doc_tests_from_file(path: &Path, tests: &mut Vec<DocTest>) -> Result<()> {
1077    let content = fs::read_to_string(path)?;
1078
1079    let mut current_item = String::from("(module)");
1080    let mut in_doc_comment = false;
1081    let mut in_code_block = false;
1082    let mut code_lines = Vec::new();
1083    let mut code_start_line = 0u32;
1084    let mut flags = DocTestFlags::default();
1085    let mut expected_output = None;
1086
1087    for (line_num, line) in content.lines().enumerate() {
1088        let line_num = line_num as u32 + 1;
1089        let trimmed = line.trim();
1090
1091        // Track current item name
1092        if let Some(rest) = trimmed
1093            .strip_prefix("fn ")
1094            .or(trimmed.strip_prefix("pub fn "))
1095            .or(trimmed.strip_prefix("task fn "))
1096            .or(trimmed.strip_prefix("pub task fn "))
1097        {
1098            if let Some(name) = rest.split(['(', '<']).next() {
1099                current_item = name.trim().to_string();
1100            }
1101        } else if let Some(rest) = trimmed
1102            .strip_prefix("struct ")
1103            .or(trimmed.strip_prefix("pub struct "))
1104        {
1105            if let Some(name) = rest.split(['<', '{']).next() {
1106                current_item = name.trim().to_string();
1107            }
1108        } else if let Some(rest) = trimmed
1109            .strip_prefix("enum ")
1110            .or(trimmed.strip_prefix("pub enum "))
1111        {
1112            if let Some(name) = rest.split(['<', '{']).next() {
1113                current_item = name.trim().to_string();
1114            }
1115        } else if let Some(rest) = trimmed
1116            .strip_prefix("trait ")
1117            .or(trimmed.strip_prefix("pub trait "))
1118        {
1119            if let Some(name) = rest.split(['<', '{']).next() {
1120                current_item = name.trim().to_string();
1121            }
1122        }
1123
1124        // Check for doc comments
1125        let doc_line = if trimmed.starts_with("///") {
1126            in_doc_comment = true;
1127            Some(trimmed.trim_start_matches("///").trim())
1128        } else if trimmed.starts_with("//!") {
1129            in_doc_comment = true;
1130            Some(trimmed.trim_start_matches("//!").trim())
1131        } else {
1132            if in_doc_comment && !trimmed.is_empty() && !trimmed.starts_with("//") {
1133                in_doc_comment = false;
1134            }
1135            None
1136        };
1137
1138        if let Some(doc) = doc_line {
1139            // Check for code block start
1140            if doc.starts_with("```") {
1141                if in_code_block {
1142                    // End of code block - create the test
1143                    if !code_lines.is_empty() {
1144                        tests.push(DocTest {
1145                            file: path.to_path_buf(),
1146                            line: code_start_line,
1147                            item: current_item.clone(),
1148                            code: code_lines.join("\n"),
1149                            expected_output: expected_output.take(),
1150                            flags,
1151                        });
1152                    }
1153                    code_lines.clear();
1154                    in_code_block = false;
1155                    flags = DocTestFlags::default();
1156                } else {
1157                    // Start of code block - parse flags from fence
1158                    in_code_block = true;
1159                    code_start_line = line_num;
1160                    flags = parse_code_fence_flags(doc);
1161                }
1162            } else if in_code_block {
1163                // Check for expected output marker
1164                if doc.starts_with("// =>") || doc.starts_with("// ->") {
1165                    expected_output = Some(
1166                        doc.trim_start_matches("// =>")
1167                            .trim_start_matches("// ->")
1168                            .trim()
1169                            .to_string(),
1170                    );
1171                } else {
1172                    code_lines.push(doc.to_string());
1173                }
1174            }
1175        }
1176    }
1177
1178    Ok(())
1179}

1181/// Parse flags from a code fence line like ```joule,ignore
1182fn parse_code_fence_flags(fence: &str) -> DocTestFlags {
1183    let mut flags = DocTestFlags::default();
1184
1185    let parts: Vec<&str> = fence
1186        .trim_start_matches('`')
1187        .split(',')
1188        .map(|s| s.trim())
1189        .collect();
1190
1191    for part in parts {
1192        match part {
1193            "ignore" => flags.ignore = true,
1194            "no_run" => flags.no_run = true,
1195            "no_compile" | "text" | "txt" => flags.no_compile = true,
1196            "compile_fail" => flags.compile_fail = true,
1197            "should_panic" => flags.should_panic = true,
1198            _ => {}
1199        }
1200    }
1201
1202    flags
1203}
